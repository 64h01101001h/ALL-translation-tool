def h(b):return hashlib.sha256(b).hexdigest()
def save(n,v):
 p=R/n;p.parent.mkdir(parents=True,exist_ok=True);p.write_text(json.dumps(v,ensure_ascii=False,indent=2)+'\n')
def git(*args):return subprocess.check_output(['git','-C',str(W),*args])
head=git('rev-parse','HEAD').decode().strip()
pins={}
for fn in ['english/241/provenance.json','english/241/executed-dependency-pins.json']:
 for a in json.loads((B/fn).read_text()):
  if 'git_commit' in a:
   raw=git('show',a['git_commit']+':'+a['git_path']);assert h(raw)==a['sha256']
   pins[a['sha256']]=(a['git_commit'],a['git_path'])
manifest=[]; seen={}
allowed={'.json','.md','.html','.stderr','.stdout','.log','.txt','.bin','.exit'}
def copy(p,rel=None,pin=None):
 p=Path(p).absolute();raw=p.read_bytes();digest=h(raw)
 if str(p) in seen:return seen[str(p)]
 if rel is None:rel='proposal-inputs/evidence/'+digest+(p.suffix if p.suffix in allowed else '.bin')
 rel=Path(rel)
 if rel.suffix not in allowed:rel=Path(str(rel)+('.txt' if rel.suffix in {'.py','.sh','.sql'} else '.bin'))
 target=R/rel;target.parent.mkdir(parents=True,exist_ok=True)
 if target.exists():assert target.read_bytes()==raw
 else:target.write_bytes(raw)
 assert target.read_bytes()==p.read_bytes()
 a={'original':str(p),'copy':str(rel),'sha256':digest,'bytes':len(raw)}
 pin=pin or (pins.get(digest) if any(z in str(p) for z in ['/governing/','/originals/']) else None)
 if pin:
  assert git('show',pin[0]+':'+pin[1])==raw
  a.update(original_git_commit=pin[0],repository_relative_path=pin[1])
 manifest.append(a);seen[str(p)]=a;return a
# Snapshot all inputs and all attempts before interpreting. Unsupported extensions remain exact bytes.
for p in sorted(B.rglob('*')):
 if not p.is_file() or p.is_relative_to(R) or '__pycache__' in p.parts:continue
 rel=p.relative_to(B)
 if rel.parts[0]=='reconciled':out=Path('reviewed-final')/Path(*rel.parts[1:])
 else:out=Path('proposal-inputs')/rel
 copy(p,out)
# Preserve original physical files separately and independently compare the proposal copies.
physical=[]
for a in json.loads((B/'reconciled/241/physical-source-reopened.json').read_text()):

 c=copy(p);physical.append({**c,'reopened_original_equals_frozen_copy':True})
save('physical-verification.json',physical)
# Immutable historical repository files are compared to Git objects, never mutable live files.
history=[]
for a in json.loads((B/'reconciled/241/original-manifest.json').read_text()):

 pin=pins.get(a['sha256']) if str(W) in a['original_path'] else None
 if pin:assert raw==git('show',pin[0]+':'+pin[1])
 elif Path(a['original_path']).is_file() and str(W) not in a['original_path']:
  # proposal-protocol is deliberately revised; original snapshot kept as its own original path.
  if Path(a['original_path']).name!='proposal-protocol.md':assert raw==Path(a['original_path']).read_bytes()
 history.append({'original':a['original_path'],'preserved_original':a['copy'],'sha256':a['sha256'],'bytes':a['bytes'],'git_pin':pin,'checked':True})
save('original-history-verification.json',history)
for rel in ['AGENTS.md','docs/ALIGNMENT_LAYER_SPEC.md','docs/alignment_briefs/RECONCILE_BRIEF.md','docs/alignment_briefs/REFUTE_BRIEF.md','docs/alignment_briefs/PROPOSE_BRIEF.md','docs/errata_register.json','data/alignment/C05_CAMPAIGN.md','data/alignment/span_head_allow.json','tools/gen_alignment_page.py','tools/build_alignment_layer.py']:
 raw=(W/rel).read_bytes();assert raw==git('show',head+':'+rel),rel
 copy(W/rel,'governing/'+rel,(head,rel))
for p in [B.parent/'semantic-review-protocol.md',B.parent/'proposal-protocol.md',B.parent/'run_generator_readonly.py',*sorted((B.parent/'policy-history').rglob('*'))]:
 if p.is_file():copy(p,'governing/campaign/'+str(p.relative_to(B.parent)))
save('original-to-copy-manifest.json',manifest)
access=[]

# Canonical source and resolver.
sp=importlib.util.spec_from_file_location('canonical_review_generator',W/'tools/gen_alignment_page.py');G=importlib.util.module_from_spec(sp);sp.loader.exec_module(G)

  sps=seg['spans'];tr=G.resolve(src['wylie'],sps,'tib',seq);byid={s['id']:s for s in sps};es=G.with_members(sps,[byid[i] for i in seg['eng_order']],src['english']);er=G.resolve(src['english'],es,'eng',seq)
  rr=[{'seq':seq,'id':s['id'],'d':s['d'],'tib':s['tib'],'eng':s['eng'],'tib_range':list(tr[s['id']]),'eng_range':list(er[s['id']]) if s['eng'] is not None else None} for s in sps]

  rec={'seq':seq,'span_count':len(sps),'null_count':sum(s['eng'] is None for s in sps),'word_pair_count':sum(s['d']==5 and s['eng'] is not None for s in sps),'spec_sha256':h(spec_raw),'spec_bytes':len(spec_raw),'body_sha256':h(proc.stdout),'body_bytes':len(proc.stdout),'generator_exit':proc.returncode,'exact_saved_body':True,'angle':angle};replays.append(rec)
  if angle=='reconciled':resolved.extend(rr)
