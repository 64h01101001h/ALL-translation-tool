def hits(text,piece,member=False):
    out=[]; i=0
    while True:
        i=text.find(piece,i) if member else G.find_word(text,piece,i)
        if i<0:return out
        out.append([i,i+len(piece)]);i+=1
def main():
    for seq,items in data.items():
        src=sources[seq]; spans=[]; decisions=[]; parents={}; expected={}; parent=None
        for i,(d,tib,eng,why,n) in enumerate(items,1):
            ident=('m' if d==7 else 'p' if d==6 else 'f' if d<5 else 'w')+str(i)
            sp={'id':ident,'d':d,'tib':tib,'eng':eng};spans.append(sp)
            if d!=7: parent=ident
            else:parents[ident]=parent
            eh=hits(src['english'],eng)
            if d!=7:
                assert len(eh)>=n,(seq,eng,n,eh)
                expected[ident]=eh[n-1]
            decisions.append({'id':ident,'d':d,'tib':tib,'eng':eng,'source_function_and_rationale':why,'all_english_word_occurrences':eh,'selected_english_occurrence_one_based':n if d!=7 else 'inside specified parent','parent':parents.get(ident)})
        en_order=sorted(expected,key=lambda ident:expected[ident][0])
        tr=G.resolve(src['wylie'],spans,'tib',seq)
        byid={x['id']:x for x in spans}
        en_spans=G.with_members(spans,[byid[x] for x in en_order],src['english'])
        er=G.resolve(src['english'],en_spans,'eng',seq)
        assert all(tuple(rng)==er[i] for i,rng in expected.items()),(seq,expected,er)
        for dec in decisions:
            i=dec['id'];dec['tib_range']=tr[i];dec['eng_range']=er[i]
            dec['tib_clause_context']=src['wylie'][max(0,tr[i][0]-90):min(len(src['wylie']),tr[i][1]+90)]
            dec['english_clause_context']=src['english'][max(0,er[i][0]-110):min(len(src['english']),er[i][1]+110)]
            dec['all_tibetan_occurrences']=hits(src['wylie'],dec['tib'],dec['d']==7)
            if dec['d']==7:
                p=dec['parent'];dec['member_proof']={'parent_tib_range':tr[p],'parent_eng_range':er[p],'unique_tib_in_parent':src['wylie'][slice(*tr[p])].count(dec['tib'])==1,'unique_eng_in_parent':src['english'][slice(*er[p])].count(dec['eng'])==1,'strictly_tighter_both':tr[i]!=tr[p] and er[i]!=er[p]}
        seg={'seq':seq,'title':{313:'possible equivalent perceptions',314:'space and the sensation of heat',315:'illumination of the path to freedom'}[seq],'spans':spans,'eng_order':en_order,'note':notes[seq]}
