def occ(text,sub):

def complement(text,ranges):
 merged=[]
 for a,b in sorted(ranges):
  if merged and a<=merged[-1][1]:merged[-1][1]=max(b,merged[-1][1])
  else:merged.append([a,b])
 out=[];pos=0
 for a,b in merged:
  if pos<a:out.append({'range':[pos,a],'text':text[pos:a]})
  pos=max(pos,b)
 if pos<len(text):out.append({'range':[pos,len(text)],'text':text[pos:]})
 return out
