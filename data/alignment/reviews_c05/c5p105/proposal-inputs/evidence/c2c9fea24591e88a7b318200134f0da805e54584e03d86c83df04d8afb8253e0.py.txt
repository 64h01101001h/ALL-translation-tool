from pathlib import Path
import json,re
OWN=Path(__file__).resolve().parents[1]
rows={r['seq']:r for r in json.loads((OWN/'evidence/source.json').read_text())}
def matches(text,piece):
    return [m.start() for m in re.finditer(re.escape(piece),text) if (not piece[0].isalnum() or m.start()==0 or not text[m.start()-1].isalnum()) and (not piece[-1].isalnum() or m.end()==len(text) or not text[m.end()].isalnum())]
def write(seq,title,plans,note,omissions):
    target=OWN/str(seq);target.mkdir(exist_ok=True)
    row=rows[seq];spans=[];decisions=[];expected={};parent=None
    for n,plan in enumerate(plans,1):
        d,t,e,ti,ei,why=plan
        sp={'id':'w'+str(n),'d':d,'tib':t,'eng':e}
        if e is None:sp['nul']=why
        spans.append(sp)
        if d==7:
            assert parent is not None
            tp=row['wylie'][slice(*parent['tib_range'])]
            ep=row['english'][slice(*parent['eng_range'])]
            assert tp.count(t)==1 and ep.count(e)==1
            a=parent['tib_range'][0]+tp.index(t);b=parent['eng_range'][0]+ep.index(e)
        else:
            th=matches(row['wylie'],t);assert len(th)>=ti,(seq,t,th,ti);a=th[ti-1]
            if e is not None:
                eh=matches(row['english'],e);assert len(eh)>=ei,(seq,e,eh,ei);b=eh[ei-1]
        record={'id':sp['id'],'d':d,'tib':t,'eng':e,'tib_range':[a,a+len(t)],'eng_range':[b,b+len(e)] if e is not None else None,'source_occurrence':ti,'english_occurrence':ei,'reason':why}
        if d!=7:parent=record if d==5 else None
        decisions.append(record)
    order=[r['id'] for r in sorted([r for r in decisions if r['eng'] is not None and r['d']!=7],key=lambda r:r['eng_range'][0])]
    spec={'course':'C05','segments':[{'seq':seq,'title':title,'spans':spans,'eng_order':order,'note':note}]}
    (target/'spec.json').write_text(json.dumps(spec,ensure_ascii=False,indent=2)+'\n')
    (target/'decisions.json').write_text(json.dumps({'producer':'Codex, English-first independent provisional proposal','source_origin':'8575edf1e0de695677194c1fde60f1d9e2c2f204 through309; not future through312 acceptance','spans':decisions,'omissions':omissions},ensure_ascii=False,indent=2)+'\n')
    (target/'errata.json').write_text('[]\n')
