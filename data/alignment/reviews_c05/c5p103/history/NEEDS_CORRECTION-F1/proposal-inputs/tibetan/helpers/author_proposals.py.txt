def hits(text,piece):
    return [m.start() for m in re.finditer(re.escape(piece),text) if (not piece[0].isalnum() or m.start()==0 or not text[m.start()-1].isalnum()) and (not piece[-1].isalnum() or m.end()==len(text) or not text[m.end()].isalnum())]
def build(seq,title,items,omissions,note):
    source=sources[seq];spans=[];decisions=[];cursor=0;parent=None
    for n,item in enumerate(items,1):
        d,tib,eng,reason,*extra=item;opts=extra[0] if extra else {};id='w'+str(n)
        sp={'id':id,'d':d,'tib':tib,'eng':eng}
        if eng is None: sp.update(cls='case',nul=reason)
        if d==7:
            lo,hi=parent['tib_range']; assert source['wylie'][lo:hi].count(tib)==1
            tpos=source['wylie'].index(tib,lo,hi)
        else:
            choices=[i for i in hits(source['wylie'],tib) if i>=cursor]
            assert choices,(seq,tib,cursor)
            tpos=choices[0];cursor=tpos+len(tib)
        epos=None
        if eng is not None:
            if d==7:
                lo,hi=parent['eng_range'];assert source['english'][lo:hi].count(eng)==1
                epos=source['english'].index(eng,lo,hi)
            elif 'anchor' in opts:
                anchor=opts['anchor']; assert source['english'].count(anchor)==1,(seq,anchor)
                epos=source['english'].index(anchor)+anchor.index(eng)
            else:
                found=hits(source['english'],eng);occ=opts.get('occ',1)
                assert len(found)>=occ,(seq,eng,found,occ)
                if len(found)>1:assert 'occ' in opts,(seq,eng,found,'explicit occurrence required')
                epos=found[occ-1]
        decision={'id':id,'d':d,'tib':tib,'eng':eng,'status':'PROVISIONAL_TIBETAN_FIRST','reason':reason,'tib_range':[tpos,tpos+len(tib)],'eng_range':[epos,epos+len(eng)] if epos is not None else None,'all_exact_tib_occurrences':hits(source['wylie'],tib),'all_exact_eng_occurrences':hits(source['english'],eng) if eng else [],'selection':opts}
        if d==5:parent=decision
        elif d!=7:parent=None
        spans.append(sp);decisions.append(decision)
    order=[d['id'] for d in sorted([d for d in decisions if d['eng'] is not None and d['d']!=7],key=lambda x:x['eng_range'][0])]
    s={'course':'C05','segments':[{'seq':seq,'title':title,'spans':spans,'eng_order':order,'note':note}]}
    out=ROOT/str(seq)
    (out/'spec.json').write_text(json.dumps(s,ensure_ascii=False,indent=2)+'\n')
    (out/'errata.json').write_text('[]\n')
    (out/'decisions.json').write_text(json.dumps({'schema':'tibetan-first-proposal-decisions-v1','producer':'Codex (gpt-6-astra), Tibetan-first independent proposal','seq':seq,'source':'../evidence/source-context-verification.json','proposals':decisions,'omissions':omissions},ensure_ascii=False,indent=2)+'\n')
    print(seq,len(spans),'spans',sum(s['eng'] is None for s in spans),'nulls',sum(s['d']==5 and s['eng'] is not None for s in spans),'d5 pairs')
