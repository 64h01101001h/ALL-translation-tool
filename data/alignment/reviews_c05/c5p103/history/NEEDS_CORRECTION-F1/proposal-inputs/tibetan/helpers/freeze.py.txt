import datetime,hashlib,json,pathlib,stat
ROOT=pathlib.Path(__file__).resolve().parents[1]
FREEZE=ROOT/'freeze-final-v1.json'
assert not FREEZE.exists(),'A frozen boundary must be explicitly superseded, never overwritten.'
def sha(b):return hashlib.sha256(b).hexdigest()
summary=json.loads((ROOT/'summary.json').read_text())
final={}
for seq in [307,308,309]:
    d=ROOT/str(seq);m=json.loads((d/'mechanical-proof.json').read_text())
    assert sha((d/'spec.json').read_bytes())==m['spec_sha256']
    assert sha((d/'body.html').read_bytes())==m['body_sha256']
    assert (d/'body.html').read_bytes()==(d/'native/attempt01/stdout.html').read_bytes()
    assert (d/'spec.json').read_bytes()==(d/'native/attempt01/stdin.json').read_bytes()
    assert (d/'native/attempt01/stderr.txt').read_bytes()==b''
    assert (d/'native/attempt01/exit.txt').read_text()=='0\n'
    assert len((d/'report.md').read_text().splitlines())==12
    assert json.loads((d/'errata.json').read_text())==[]
    for r in json.loads((d/'evidence-references.json').read_text())['references']:
        assert sha((d/r['path']).read_bytes())==r['sha256']
    final[str(seq)]={'spans':m['spans'],'nulls':m['nulls'],'d5_nonnull':m['d5_nonnull'],'body_bytes':m['body_bytes'],'body_sha256':m['body_sha256'],'spec_sha256':m['spec_sha256'],'errata':0,'canonical_exit':0,'stderr_bytes':0}

print(json.dumps({'freeze_path':str(FREEZE),'schema':proof['schema'],'sha256':sha(FREEZE.read_bytes()),'file_count':len(files),'file_bytes':proof['file_bytes'],'segments':final},indent=2))
