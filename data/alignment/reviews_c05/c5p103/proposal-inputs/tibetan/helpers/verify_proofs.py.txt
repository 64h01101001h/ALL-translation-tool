class SourceParser(HTMLParser):
    def __init__(self):
        super().__init__(convert_charrefs=True);self.stack=[];self.plain={'tib':'','eng':''};self.ranges={'tib':{},'eng':{}}
    def handle_starttag(self,tag,attrs):
        a=dict(attrs);side=next((f['side'] for f in reversed(self.stack) if f['side']),None)
        if tag=='div' and a.get('class') in ['tib','eng']:side=a['class']
        lab=a.get('data-l');active=bool(lab and lab.endswith('S')) or any(f['source'] for f in self.stack)
        self.stack.append({'tag':tag,'side':side,'source':active,'nul':a.get('class')=='nul','lab':lab,'start':len(self.plain[side]) if side else None})
    def handle_data(self,data):
        if self.stack and any(f['source'] for f in self.stack) and not any(f['nul'] for f in self.stack):
            side=self.stack[-1]['side']
            if side:self.plain[side]+=data
    def handle_endtag(self,tag):
        assert self.stack and self.stack[-1]['tag']==tag,(tag,self.stack[-1] if self.stack else None)
        f=self.stack.pop()
        if f['lab'] and f['side']:self.ranges[f['side']][f['lab']]=[f['start'],len(self.plain[f['side']])]
