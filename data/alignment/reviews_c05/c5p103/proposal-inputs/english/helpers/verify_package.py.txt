class BodyParser(HTMLParser):
    def __init__(self):
        super().__init__(convert_charrefs=True);self.side=None;self.divs=[];self.spans=[];self.text={'tib':'','eng':''};self.ranges={'tib':{},'eng':{}};self.active=0;self.skip=0
    def handle_starttag(self,tag,attrs):
        a=dict(attrs)
        if tag=='div':
            self.divs.append(self.side)
            if a.get('class') in ('tib','eng'):self.side=a['class']
        if tag=='span':
            label=a.get('data-l');nul=a.get('class')=='nul';root=label and label.endswith('S')
            self.spans.append((label,nul,root,self.side,len(self.text[self.side]) if self.side else 0,a.get('data-d')))
            if root:self.active+=1
            if nul:self.skip+=1
    def handle_data(self,data):
        if self.active and not self.skip and self.side:self.text[self.side]+=data
    def handle_endtag(self,tag):
        if tag=='span':
            label,nul,root,side,start,depth=self.spans.pop()
            if label and side:self.ranges[side][label]={'range':[start,len(self.text[side])],'depth':int(depth)}
            if nul:self.skip-=1
            if root:self.active-=1
        if tag=='div':self.side=self.divs.pop()
