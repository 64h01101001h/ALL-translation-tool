from pathlib import Path

pin=json.loads((P.parent/'proposal-baseline.json').read_text())
original_connect=sqlite3.connect
def readonly(database,*args,**kwargs):
 assert str(database)==str(W/'build/hgm_spine_v27_2.db'),str(database)

 return original_connect(Path(database).as_uri()+'?mode=ro',*args,**kwargs)

 print(json.dumps(dict(status='PASS',segments=out),ensure_ascii=False,indent=2))
else:raise RuntimeError('unknown mode')
