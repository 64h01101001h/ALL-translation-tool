import datetime,hashlib,json,os,pathlib,subprocess,sys,time
o=pathlib.Path(sys.argv[1]).resolve();o.mkdir(parents=True)
argv=sys.argv[2:]; env=dict(os.environ,PYTHONDONTWRITEBYTECODE='1'); t=time.perf_counter()
(o/'stdin').write_bytes(b'')
r=subprocess.run(argv,input=b'',stdout=subprocess.PIPE,stderr=subprocess.PIPE,env=env)
for n,b in [('stdout',r.stdout),('stderr',r.stderr)]: (o/n).write_bytes(b)
(o/'exit').write_text(str(r.returncode)+'\n');(o/'execution.json').write_text(json.dumps(dict(argv=argv,cwd=os.getcwd(),environment_overrides={'PYTHONDONTWRITEBYTECODE':'1'},seconds=time.perf_counter()-t,exit=r.returncode,stdout_bytes=len(r.stdout),stderr_bytes=len(r.stderr),utc=datetime.datetime.now(datetime.timezone.utc).isoformat()),indent=2)+'\n')
print(r.stdout.decode());print(r.stderr.decode(),file=sys.stderr);sys.exit(r.returncode)
