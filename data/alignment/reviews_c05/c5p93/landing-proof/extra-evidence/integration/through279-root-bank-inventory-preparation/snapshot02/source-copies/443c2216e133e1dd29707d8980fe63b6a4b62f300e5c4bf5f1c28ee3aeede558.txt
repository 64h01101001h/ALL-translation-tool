"""Root identity gate AFTER reading the independent final semantic review.

Current277–279 schema only. This checker does not supply a semantic judgment,
rerun SQLite queries, or regenerate canonical bodies. All actual evidence is
read in full and retained by the review freeze; new Git reads are recorded.
Usage: python3 -B verify-final-semantic-review-root.py REVIEW_FREEZE_SHA256
"""
from pathlib import Path
import datetime, hashlib, json, os, subprocess, sys

B=Path(__file__).resolve().parent
R=B/'semantic-review'
A=B.parent
W=A.parent/'campaign-worktree'
O=B/'root-final-semantic-review-verification'
assert not O.exists(), 'Preserve every previous root attempt before rerunning'
O.mkdir()
HEAD='f114fcb7a5a899b939ffc948ae9fdbbb746541fa'
SEQS=[277,278,279]
def pin(p):
    raw=p.read_bytes()
    return dict(bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest())
def read(p):return json.loads(p.read_bytes())
def save(p,v):p.write_text(json.dumps(v,ensure_ascii=False,indent=2)+'\n')
def call(argv,label):
    folder=O/label;folder.mkdir()
    start=datetime.datetime.now(datetime.timezone.utc).isoformat()
    run=subprocess.run(argv,cwd=W,input=b'',capture_output=True,env={**os.environ,'GIT_OPTIONAL_LOCKS':'0','PYTHONDONTWRITEBYTECODE':'1','PYTHONOPTIMIZE':'0'})
    for name,raw in [('stdout.bin',run.stdout),('stderr.bin',run.stderr),('stdin.bin',b''),('exit.txt',f'{run.returncode}\n'.encode())]:
        (folder/name).write_bytes(raw)
    save(folder/'execution.json',dict(argv=argv,cwd=str(W),started_at=start,finished_at=datetime.datetime.now(datetime.timezone.utc).isoformat(),exit=run.returncode,stdout=pin(folder/'stdout.bin'),stderr=pin(folder/'stderr.bin'),environment_overrides={'GIT_OPTIONAL_LOCKS':'0','PYTHONDONTWRITEBYTECODE':'1','PYTHONOPTIMIZE':'0'}))
    assert run.returncode==0 and not run.stderr,(label,run.returncode)
    return run.stdout

    supported={'.json','.md','.html','.stderr','.stdout','.log','.txt','.bin','.exit'}
    assert all(Path(path).suffix.lower() in supported for path in actual), 'Every frozen byte must survive canonical review copy'
    meta=read(R/'review.json')
    assert meta['verdict']==meta['spec_verdict']==meta['quality_verdict']=='APPROVE'
    assert meta['open_findings']==[]
    assert meta['status']=='APPROVE' and meta['baseline_commit']==HEAD
    assert meta['report']==pin(R/'review.md')
    assert meta['checklist'] and all(r['status']=='APPROVE' for r in meta['checklist'])
    assert [r['seq'] for r in meta['segments']]==SEQS
    assert freeze['spec_verdict']==freeze['quality_verdict']=='APPROVE'
    assert read(R/'bank-ready-errata.json')==[]
    assert read(R/'span-head-allow-needed.json')=={}
    for category in ['tibetan','english','reconciled']:
        for seq in SEQS:
            folder=B/category/str(seq)
            f=read(folder/'freeze.json')
            values=f['files']
            if category=='tibetan':
                assert isinstance(values,list) and all(set(r)=={'path','bytes','sha256'} for r in values)

                expected=values
