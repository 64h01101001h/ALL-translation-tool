from pathlib import Path
import datetime,json,os,subprocess,sys
D=Path(__file__).resolve().parent; R=D/'adaptation-run01';R.mkdir()
argv=[sys.executable,'-B',str(D/'adapt-concrete.py')]
(R/'command.json').write_text(json.dumps(dict(argv=argv,cwd=str(Path.cwd()),environment_overrides={'PYTHONDONTWRITEBYTECODE':'1','PYTHONOPTIMIZE':'0'},utc=datetime.datetime.now(datetime.timezone.utc).isoformat()),indent=2)+'\n');(R/'stdin').write_bytes(b'')
p=subprocess.run(argv,input=b'',stdout=subprocess.PIPE,stderr=subprocess.PIPE,env=dict(os.environ,PYTHONDONTWRITEBYTECODE='1',PYTHONOPTIMIZE='0'))
(R/'stdout').write_bytes(p.stdout);(R/'stderr').write_bytes(p.stderr);(R/'exit').write_text(str(p.returncode)+'\n')
print(p.stdout.decode());print(p.stderr.decode());print('EXIT',p.returncode)
