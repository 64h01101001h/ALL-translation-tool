import json,hashlib,os,datetime

I=D.parents[1]; ROOT=I.parents[1]; B=ROOT/'campaign-artifacts/C05-277-279'; W=ROOT/'campaign-worktree';L=B/'landing'
def pin(b):return {'bytes':len(b),'sha256':hashlib.sha256(b).hexdigest()}
def dump(p,x):p.write_text(json.dumps(x,indent=2)+'\n')
records=[]
files=[p for p in L.iterdir() if p.is_file() and p.name!='previous-main-csv-seed.bin']

 b=p.read_bytes(); t=D/'inputs'/p.relative_to(B)

 with t.open('xb') as f:f.write(b)

 records.append({'original':str(p),'resolved_original':str(p.resolve()),'symlink':os.readlink(p) if p.is_symlink() else None,'copy':str(t.relative_to(D)),**pin(b)})
callers=[]
for d in sorted((B/'landing-runs').iterdir()):

assert len(payload)==12
build=json.loads((L/'canonical-builds-once.json').read_bytes())
for prefix,rec in zip(['build-layer','build-view'],build['runs']):
 assert rec['exit']==0
