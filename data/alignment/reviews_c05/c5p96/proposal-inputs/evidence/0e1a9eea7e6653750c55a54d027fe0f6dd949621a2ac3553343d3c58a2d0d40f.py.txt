import json,pathlib,re
E=pathlib.Path(__file__).parent
A=E.parents[1]
rows={r['seq']:r for r in json.loads((E/'original-context-full-rows.json').read_text())}
def make(seq,title,pairs,note):
    ss=[];choices=[]
    for i,p in enumerate(pairs,1):
        d,tib,eng,*opt=p;sp={'id':'w'+str(i),'d':d,'tib':tib,'eng':eng}
        if eng is None:sp.update(cls='neg',nul=opt[0]);choices.append({'id':sp['id'],'null_reason':opt[0]})
        else:
            occ=opt[0] if opt else 0
            hits=[m.start() for m in re.finditer(r'(?<![A-Za-z0-9])'+re.escape(eng)+r'(?![A-Za-z0-9])',rows[seq]['english'])]
            assert len(hits)>occ,(seq,tib,eng,occ,hits)
            choices.append({'id':sp['id'],'english':eng,'all_whole_word_offsets':hits,'chosen_occurrence_zero_based':occ,'chosen_offset':hits[occ]})
        ss.append(sp)
    order=[x['id'] for x in sorted([x for x in choices if 'chosen_offset' in x],key=lambda x:x['chosen_offset'])]
    spec={'course':'C05','segments':[{'seq':seq,'title':title,'spans':ss,'eng_order':order,'note':'PROVISIONAL machine alignment. Actual producer: Codex (gpt-6-astra), independent English-first proposal; no human semantic approval and no hgm_gloss change. '+note}]}
    target=A/str(seq)
    (target/'spec.json').write_text(json.dumps(spec,ensure_ascii=False,indent=2)+'\n')
    (target/'occurrence-intent.json').write_text(json.dumps(choices,ensure_ascii=False,indent=2)+'\n')
    (target/'errata.json').write_text('[]\n')
    print(seq,len(ss),sum(s['d']==5 for s in ss),sum(s['eng'] is None for s in ss))
