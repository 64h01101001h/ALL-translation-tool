import hashlib,json,pathlib
p=pathlib.Path(__file__).resolve().parent
assert not (p/'freeze.json').exists()
rows=[]
for f in sorted(p.rglob('*')):
    if f.is_file():
        b=f.read_bytes()
        rows.append(dict(path=str(f.relative_to(p)),bytes=len(b),sha256=hashlib.sha256(b).hexdigest()))
freeze=dict(scope='Independent English-first source preparation only; no proposal/acceptance/baseline',files=rows,file_count=len(rows),total_bytes=sum(r['bytes'] for r in rows))
b=(json.dumps(freeze,ensure_ascii=False,indent=2)+'\n').encode()
(p/'freeze.json').write_bytes(b)
for r in rows:
    b2=(p/r['path']).read_bytes()
    assert len(b2)==r['bytes'] and hashlib.sha256(b2).hexdigest()==r['sha256']
print(json.dumps(dict(status='DONE',file_count=len(rows),total_bytes=freeze['total_bytes'],freeze_sha256=hashlib.sha256(b).hexdigest(),path=str(p/'freeze.json'))))
