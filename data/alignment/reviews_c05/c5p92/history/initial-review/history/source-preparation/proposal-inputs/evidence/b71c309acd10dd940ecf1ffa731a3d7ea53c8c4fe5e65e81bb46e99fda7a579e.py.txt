import datetime, hashlib, json, os, pathlib, subprocess, sys, time
p=pathlib.Path(__file__).resolve().parent
d=p/'run01';d.mkdir()
argv=[sys.executable,'-B',str(p/'capture.py')]
env=dict(os.environ,PYTHONDONTWRITEBYTECODE='1')
(d/'stdin').write_bytes(b'')
start=datetime.datetime.now(datetime.timezone.utc).isoformat(); clock=time.perf_counter()
r=subprocess.run(argv,input=b'',stdout=subprocess.PIPE,stderr=subprocess.PIPE,cwd=p,env=env)
(d/'stdout').write_bytes(r.stdout);(d/'stderr').write_bytes(r.stderr);(d/'exit').write_text(str(r.returncode)+'\n')
(d/'execution.json').write_text(json.dumps(dict(argv=argv,cwd=str(p),environment_overrides={'PYTHONDONTWRITEBYTECODE':'1'},started_utc=start,seconds=time.perf_counter()-clock,exit=r.returncode,stdin_bytes=0,stdout_bytes=len(r.stdout),stderr_bytes=len(r.stderr),stdout_sha256=hashlib.sha256(r.stdout).hexdigest(),stderr_sha256=hashlib.sha256(r.stderr).hexdigest()),indent=2)+'\n')
print(r.stdout.decode());print(r.stderr.decode(),file=sys.stderr);sys.exit(r.returncode)
