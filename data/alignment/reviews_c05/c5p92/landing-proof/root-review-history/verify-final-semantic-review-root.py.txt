def pin(raw):return dict(bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest())
def js(p):return json.loads(p.read_bytes())
def save(p,x):p.write_text(json.dumps(x,ensure_ascii=False,indent=2)+'\n')
commands=[]; objects={}
def git(*args):
    q=O/'git'/str(len(commands));q.mkdir(parents=True)
    record=dict(argv=['git',*args],cwd=str(W),environment_overrides={'GIT_OPTIONAL_LOCKS':'0'})
    save(q/'command.json',record);(q/'stdin.bin').write_bytes(b'')
    r=subprocess.run(record['argv'],cwd=W,env=dict(os.environ,GIT_OPTIONAL_LOCKS='0'),input=b'',capture_output=True)
    (q/'stdout.bin').write_bytes(r.stdout);(q/'stderr.bin').write_bytes(r.stderr);(q/'exit.txt').write_text(str(r.returncode)+'\n')
    commands.append(record);assert r.returncode==0 and not r.stderr
    return r.stdout
