def save(name,value):

def retain(source,name):
 before=source.stat();raw=source.read_bytes();after=source.stat()
 fields=lambda s:(s.st_dev,s.st_ino,s.st_size,s.st_mtime_ns,s.st_ctime_ns)
 assert fields(before)==fields(after)
 dest=O/'sources'/name;dest.parent.mkdir(parents=True,exist_ok=True)
 with dest.open('xb') as f:f.write(raw)

 bindings.append(dict(original=str(source),copy=str(dest.relative_to(O)),observed_utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),**pin(raw)))
 return raw
