records = []
def pin(raw): return dict(bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest())
def save(name, value):
    p = O / name
    p.parent.mkdir(parents=True, exist_ok=True)
    with p.open('x') as f: json.dump(value, f, ensure_ascii=False, indent=2); f.write('\n')
def copy(p, name):
    raw = p.read_bytes()
    d = O / 'inputs' / name
    d.parent.mkdir(parents=True, exist_ok=True)
    with d.open('xb') as f: f.write(raw)
    assert raw == p.read_bytes() == d.read_bytes()
    records.append(dict(original=str(p), resolved=str(p.resolve()), symlink=os.readlink(p) if p.is_symlink() else None, copy=str(d.relative_to(O)), **pin(raw)))
    return raw

proofs = []
for name, changes in adaptations.items():
    old = methods[name].decode()
    new = old
    for before, after, count in changes:
        assert new.count(before) == count, (name, before)
        new = new.replace(before, after)
    inv = new
    for before, after, count in reversed(changes):
        assert inv.count(after) == count, (name, after)
        inv = inv.replace(after, before)
    assert inv == old
    ast.parse(new)
    target = 'bank_root_proofs_through282.py.txt' if name.startswith('bank_') else name
    path = O / 'known-parameter-candidates' / target
    path.parent.mkdir(exist_ok=True)
    with path.open('x') as f: f.write(new)
    diff = ''.join(difflib.unified_diff(old.splitlines(True), new.splitlines(True), fromfile=name, tofile=target))
    reverse = ''.join(difflib.unified_diff(new.splitlines(True), old.splitlines(True), fromfile=target, tofile=name))
    for suffix, data in [('diff', diff), ('inverse.diff', reverse)]:
        with (O / (target + '.' + suffix)).open('x') as f: f.write(data)
    # Function bodies stay byte-for-AST exact; only module-level constant values change.
    funcs = lambda text: [ast.dump(n, include_attributes=False) for n in ast.parse(text).body if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef))]
    if name.startswith('bank_'):
        # The current batch name and expected coverage reside inside main(). Apply inverse before comparing.
        assert funcs(inv) == funcs(old)
    proofs.append(dict(source=name, candidate=str(path.relative_to(O)), original=pin(old.encode()), proposed=pin(new.encode()), replacements=[dict(old=x, new=y, count=n) for x,y,n in changes], inverse_exact=True, complete_diff_exact=True, syntax='ast.parse only', executed=False))
