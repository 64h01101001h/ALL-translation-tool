def ro_connect(p, *a, **kw):
    check('SQLite connection read-only', 'mode=rw' not in str(p))
    if not str(p).startswith('file:'): p = Path(p).resolve().as_uri() + '?mode=ro'
    elif 'mode=ro' not in str(p): raise AssertionError('Non-readonly URI')
    kw['uri'] = True
    c=real_connect(p, *a, **kw)
    c.execute('PRAGMA query_only=ON')
    check('SQLite query_only enabled',c.execute('PRAGMA query_only').fetchone()[0]==1)
    return c
class Capture(io.StringIO):

    def close(self): pass
    def bytes(self): return self.getvalue().encode(self.codec)
def virtual_open(p, mode='r', buffering=-1, encoding=None, *a, **kw):
    if isinstance(p,int): return real_io_open(p,mode,buffering,encoding,*a,**kw)
    p = str(p)
    if any(c in mode for c in 'wax+'):
        check('Builder write captured', rel(p) is not None, p)
        f = Capture(encoding); outputs[p] = f; return f
    if p in outputs: b = outputs[p].bytes()
    elif p in overrides: b = overrides[p]
    elif rel(p) in tracked: b = blob(rel(p))
    else: return real_io_open(p, mode, buffering, encoding, *a, **kw)
    return io.BytesIO(b) if 'b' in mode else io.StringIO(b.decode(encoding or 'utf-8'), newline=kw.get('newline'))
def virtual_exists(p):
    r = rel(p)
    return True if str(p) in outputs or str(p) in overrides or r in tracked else real_exists(p)
def virtual_replace(a, b): outputs[str(b)] = outputs.pop(str(a))
