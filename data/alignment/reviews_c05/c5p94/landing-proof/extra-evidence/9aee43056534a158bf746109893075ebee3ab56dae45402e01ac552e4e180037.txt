def pin(p):
    raw = Path(p).read_bytes()
    return dict(bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest())

def read(p):
    return json.loads(Path(p).read_bytes())

def save(name, value):
    (O / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

def call(argv, label):
    folder = O / label
    folder.mkdir()
    start = datetime.datetime.now(datetime.timezone.utc).isoformat()
    env = dict(os.environ, GIT_OPTIONAL_LOCKS='0', PYTHONDONTWRITEBYTECODE='1', PYTHONOPTIMIZE='0')
    result = subprocess.run(argv, cwd=W, env=env, input=b'', capture_output=True)
    for name, raw in [('stdin.bin', b''), ('stdout.bin', result.stdout), ('stderr.bin', result.stderr), ('exit.txt', (str(result.returncode) + '\n').encode())]:
        (folder / name).write_bytes(raw)
    (folder / 'execution.json').write_text(json.dumps(dict(argv=argv, cwd=str(W), start=start, end=datetime.datetime.now(datetime.timezone.utc).isoformat(), exit=result.returncode, streams={n: pin(folder / n) for n in ['stdin.bin', 'stdout.bin', 'stderr.bin']}), indent=2) + '\n')
    assert result.returncode == 0 and not result.stderr
    return result.stdout
