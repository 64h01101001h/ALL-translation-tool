"""Root fidelity checks on completed reconciliations, never semantic approval.

Usage: PYTHONDONTWRITEBYTECODE=1 python3 verify_reconciled_bodies.py 214 215 216
Replays the unchanged canonical generator and its actual occurrence resolver.
All evidence is written outside the accepted repository.
"""
from collections import Counter
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import shutil
import sqlite3
import subprocess
import sys

art = Path(__file__).resolve().parent
wt = art.parent / 'campaign-worktree'
seqs = list(map(int, sys.argv[1:]))
assert len(seqs) == 3 and seqs == list(range(seqs[0], seqs[0] + 3))
batch = art / f'C05-{seqs[0]}-{seqs[-1]}'
latest = batch / 'root-reconciliation-verification'
attempts = latest / 'attempts'
attempts.mkdir(parents=True, exist_ok=True)
out = attempts / f'{len(list(attempts.iterdir())) + 1:02d}'
out.mkdir(exist_ok=False)
sys.path.insert(0, str(wt / 'tools'))
definition = importlib.util.spec_from_file_location('canonical_generator', wt / 'tools/gen_alignment_page.py')
generator = importlib.util.module_from_spec(definition)
definition.loader.exec_module(generator)
db = sqlite3.connect((wt / 'build/hgm_spine_v27_2.db').as_uri() + '?mode=ro', uri=True)
db.row_factory = sqlite3.Row
db.execute('PRAGMA query_only=ON')
report, resolved = [], []
for seq in seqs:
    directory = batch / 'reconciled' / str(seq)
    raw = (directory / 'spec.json').read_bytes()
    spec = json.loads(raw)
    assert spec['course'] == 'C05' and [s['seq'] for s in spec['segments']] == [seq]
    segment = spec['segments'][0]
    source = dict(db.execute('SELECT * FROM corpus_segments WHERE course=? AND seq=?', ('C05', seq)).fetchone())
    run = subprocess.run([sys.executable, str(art / 'run_generator_readonly.py')], cwd=wt,
        input=raw, capture_output=True, env=dict(os.environ, PYTHONDONTWRITEBYTECODE='1'))
    for extension, data in (('stdout', run.stdout), ('stderr', run.stderr), ('exit', (str(run.returncode) + '\n').encode())):
        (out / f'{seq}.{extension}').write_bytes(data)
    assert run.returncode == 0 and not run.stderr and run.stdout == (directory / 'body.html').read_bytes(), seq
    spans = segment['spans']
    tib_ranges = generator.resolve(source['wylie'], spans, 'tib', seq)
    by_id = {s['id']: s for s in spans}
    assert segment.get('eng_order'), 'This helper requires the explicit English order; do not guess it.'
    english_spans = generator.with_members(spans, [by_id[x] for x in segment['eng_order']], source['english'])
    eng_ranges = generator.resolve(source['english'], english_spans, 'eng', seq)
    for span in spans:
        resolved.append(dict(seq=seq, id=span['id'], d=span['d'], tib=span['tib'], eng=span['eng'],
            tib_range=list(tib_ranges[span['id']]),
            eng_range=list(eng_ranges[span['id']]) if span['eng'] is not None else None))
    report.append(dict(seq=seq, spans=len(spans), depths=dict(Counter(s['d'] for s in spans)),
        nulls=sum(s['eng'] is None for s in spans), nonnull_d5=sum(s['d']==5 and s['eng'] is not None for s in spans),
        errata_candidates=len(json.loads((directory / 'errata.json').read_text())),
        generator_exit=run.returncode, body_bytes=len(run.stdout), exact_saved_body=True,
        body_sha256=hashlib.sha256(run.stdout).hexdigest(), spec_sha256=hashlib.sha256(raw).hexdigest()))
db.close()
(out / 'resolved-spans.json').write_text(json.dumps(resolved, ensure_ascii=False, indent=2) + '\n')
(out / 'report.json').write_text(json.dumps(report, indent=2) + '\n')
for path in out.iterdir():
    shutil.copy2(path, latest / path.name)
print(json.dumps(report, indent=2))
