#!/usr/bin/env python3
"""Run unchanged canonical generator with its sole DB connection forced URI/read-only."""
import sys, sqlite3, runpy
from pathlib import Path
worktree = Path(sys.argv[1]).resolve()
target = worktree / 'tools' / 'gen_alignment_page.py'
real_connect = sqlite3.connect
expected = (worktree / 'build' / 'hgm_spine_v27_2.db').resolve()
def read_only_connect(database, *args, **kwargs):
    if Path(database).resolve() != expected:
        raise RuntimeError('Unexpected database requested by generator: ' + str(database))
    kwargs['uri'] = True
    return real_connect(expected.as_uri() + '?mode=ro', *args, **kwargs)
sqlite3.connect = read_only_connect
sys.argv = [str(target)]
runpy.run_path(str(target), run_name='__main__')
