def hits(text,piece,sub=False):
 out=[]
 for m in re.finditer(re.escape(piece),text):
  i,j=m.span()
  if sub or ((not piece[0].isalnum() or i==0 or not text[i-1].isalnum()) and (not piece[-1].isalnum() or j==len(text) or not text[j].isalnum())): out.append([i,j])
 return out

 spans=[]; intended=[]
 for item in items:
  tib,eng,*args=item; d=args[0] if args else 5; tn=args[1] if len(args)>1 else 1; en=args[2] if len(args)>2 else 1
  sub=args[3] if len(args)>3 else False
  sp={'id':'w'+str(len(spans)+1),'d':d,'tib':tib,'eng':eng}
  if sub: sp['subword']=True
  if eng is None: sp.update(cls='case',nul='Agentive case marks the English subject; no separate English case exponent.')
  tr=hits(rows[seq]['wylie'],tib,sub)[tn-1]; er=hits(rows[seq]['english'],eng,sub)[en-1] if eng is not None else None

 out=BASE/str(seq)
 for name,obj in [('spec.json',spec),('intended-spans.json',intended),('decisions.json',decisions),('errata.json',[])]: (out/name).write_text(json.dumps(obj,ensure_ascii=False,indent=2)+'\n')
