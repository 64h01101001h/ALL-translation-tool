querycache={}
def query(label,sql,params=()):
    key=(sql,tuple(params))
    if key in querycache:
        q=querycache[key]; q['labels'].append(label)
        return [rowpool[r]['row'] for r in q['row_refs']]
    rows=[dict(r) for r in con.execute(sql,params)]
    table='corpus_segments' if 'FROM corpus_segments' in sql else 'entries'
    refs=[]
    for row in rows:
        ref=table+':'+str(row['id'])
        record={'table':table,'row':row}
        assert ref not in rowpool or rowpool[ref]==record
        rowpool[ref]=record; refs.append(ref)
    q={'labels':[label],'sql':sql,'parameters':list(params),'count':len(rows),'row_refs':refs}
    querycache[key]=q; querylog.append(q)
    return rows
