import json,hashlib,datetime
from pathlib import Path
BASE=Path(__file__).resolve().parent.parent
for seq in [280,281,282]:
    directory=BASE/str(seq)
    assert not (directory/'freeze.json').exists(),'Never silently overwrite a frozen boundary.'
    files=[]
    for path in sorted(directory.rglob('*')):
        if path.is_file() and path!=directory/'freeze.json':
            data=path.read_bytes();files.append({'path':path.relative_to(directory).as_posix(),'sha256':hashlib.sha256(data).hexdigest(),'bytes':len(data)})
    freeze={'segment':f'C05:{seq}','angle':'English-first','producer':'Codex (GPT-6 Astra)','status':'PROVISIONAL, unreviewed original proposal','frozen_at_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'files':files}
    dest=directory/'freeze.json';dest.write_text(json.dumps(freeze,indent=2)+'\n')
    print(seq,len(files),'files',hashlib.sha256(dest.read_bytes()).hexdigest())
for seq in [280,281,282]:
    directory=BASE/str(seq);freeze=json.loads((directory/'freeze.json').read_text())
    assert {p.relative_to(directory).as_posix() for p in directory.rglob('*') if p.is_file() and p.name!='freeze.json'}=={f['path'] for f in freeze['files']}
    for f in freeze['files']:
        b=(directory/f['path']).read_bytes();assert len(b)==f['bytes'] and hashlib.sha256(b).hexdigest()==f['sha256']
    print(seq,'frozen manifest verified')
