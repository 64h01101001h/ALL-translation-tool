"""Capture immutable source preparation; never create proposals or approvals.

Usage: python3 -B capture_future_source_batch.py FIRST LAST TERMS_JSON
The terms file is a reviewed list of exact exploratory Wylie lookup strings.
"""
from pathlib import Path
import datetime
import gzip
import hashlib
import json
import re
import sqlite3
import sys

art = Path(__file__).resolve().parent
wt = art.parent / 'campaign-worktree'
first, last = map(int, sys.argv[1:3])
assert last == first + 2
b = art / f'C05-{first}-{last}'
b.mkdir(exist_ok=True)
assert not (b / 'source.json').exists(), 'Preserve completed or failed earlier work before any retry.'
terms_path = Path(sys.argv[3]).resolve()
terms = json.loads(terms_path.read_text())
assert isinstance(terms, list) and all(isinstance(t, str) and t for t in terms)
assert len(terms) == len(set(terms))

def pin(p):
    raw = p.read_bytes()
    return dict(path=str(p), bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest())

def write(name, obj):
    (b / name).write_text(json.dumps(obj, ensure_ascii=False, indent=2) + '\n')

spine = wt / 'build/hgm_spine_v27_2.db'
master_path = wt / 'data/hgm_dictionary_v27_2.json.gz'
corpus_path = wt / 'data/full_parallel_corpus_v32.json.gz'
master = json.loads(gzip.decompress(master_path.read_bytes()))['unified_entries']
corpus = json.loads(gzip.decompress(corpus_path.read_bytes()))
course = [x for x in corpus if x['course'] == 'C05']
db = sqlite3.connect(spine.as_uri() + '?mode=ro', uri=True)
db.row_factory = sqlite3.Row
db.execute('PRAGMA query_only=ON')
rows = [dict(r) for r in db.execute('SELECT * FROM corpus_segments WHERE course=? AND seq BETWEEN ? AND ? ORDER BY seq', ('C05', first - 6, last + 6))]
assert [r['seq'] for r in rows] == list(range(first - 6, last + 7))
for r in rows:
    assert json.loads(r['raw']) == course[r['seq'] - 1]
    for key in ('course', 'wylie', 'acip', 'english'):
        assert r[key] == json.loads(r['raw'])[key]
selected = [r for r in rows if first <= r['seq'] <= last]
trim = lambda r: {k: r[k] for k in ('id', 'course', 'seq', 'acip', 'wylie', 'english')}
write('source.json', [trim(r) for r in selected])
write('context.json', [trim(r) for r in rows])
write('root-preparation-original-rows.json', rows)

sql = 'SELECT DISTINCT e.* FROM entries e LEFT JOIN entry_variants v ON v.entry_id=e.id WHERE e.wylie=? OR v.wylie=? ORDER BY e.id'
lexical = []
for term in terms:
    matches = [dict(r) for r in db.execute(sql, (term, term))]
    full_master = []
    for r in matches:
        raw = json.loads(r['raw'])
        assert raw == master[r['id'] - 1], ('master ID mapping', r['id'])
        full_master.append({'entry_id': r['id'], 'master_index': r['id'] - 1, 'entry': raw})
    lexical.append(dict(term=term, sql=sql, parameters=[term, term], count=len(matches), rows=matches, full_master_entries=full_master))
write('root-lexical-preparation.json', lexical)
parallels = []
for r in selected:
    for field in ('acip', 'wylie', 'english'):
        sql = f'SELECT * FROM corpus_segments WHERE {field}=? ORDER BY course,seq'
        matches = [dict(x) for x in db.execute(sql, (r[field],))]
        parallels.append(dict(seq=r['seq'], field=field, sql=sql, parameters=[r[field]], count=len(matches), rows=matches))
write('root-parallel-preparation.json', parallels)
for query in lexical + parallels:
    actual = [dict(r) for r in db.execute(query['sql'], query['parameters'])]
    assert actual == query['rows'] and len(actual) == query['count']
db.close()

physical = Path('/Users/adamderickandrade/Desktop/HGM DICTIONARY - TRANSLATION APP/To DO/Formal Courses/ACI Readings In Roman ASCII/C05ReadingASCII.txt')
raw = physical.read_bytes()
extents = []
fold = lambda x: re.sub(rb'[\t\n\r\f\v ]+', b' ', x).strip()
for r in selected:
    for field in ('acip', 'english'):
        value = r[field].encode('ascii')
        pattern = rb'[\t\n\r\f\v ]+'.join(re.escape(t) for t in value.split())
        matches = list(re.finditer(pattern, raw))
        assert len(matches) == 1, (r['seq'], field, 'full physical matches', len(matches))
        m = matches[0]
        assert fold(m.group()) == fold(value)
        extents.append(dict(seq=r['seq'], field=field, start=m.start(), bytes=len(m.group()), sha256=hashlib.sha256(m.group()).hexdigest(), original_text=m.group().decode('ascii'), ascii_whitespace_collapse_equals_spine=True))
write('root-preparation-proof.json', dict(timestamp=datetime.datetime.now(datetime.timezone.utc).isoformat(), status='SOURCE PREPARATION ONLY; no baseline, proposal, semantic approval or erratum', spine=pin(spine), master=pin(master_path), corpus=pin(corpus_path), source=pin(b/'source.json'), context=pin(b/'context.json'), terms=pin(terms_path), helper=pin(Path(__file__).resolve()), physical=pin(physical), physical_extents=extents, full_context_rows=len(rows), lexical_queries=len(lexical), full_entry_occurrences=sum(x['count'] for x in lexical), parallel_queries=len(parallels), all_complete_query_replays_exact=True, original_master_and_course_objects_equal=True, note='Only selected ASCII extents were decoded. No complete-file encoding claim. Full material lexical and parallel results still require model inspection before preparation freeze. Digital identity does not establish independent publication.'))
print(json.dumps(dict(source=pin(b/'source.json'), context_rows=len(rows), lexical_queries=len(lexical), full_entry_occurrences=sum(x['count'] for x in lexical), parallel_queries=len(parallels), physical_extents=len(extents))))
