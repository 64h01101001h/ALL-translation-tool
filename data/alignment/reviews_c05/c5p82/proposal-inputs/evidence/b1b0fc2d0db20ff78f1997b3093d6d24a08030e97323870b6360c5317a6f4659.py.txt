 def add(self,t,e,d=5,n=1,reason='',**kw):
  ident=('p' if d==6 else 'w')+str(len(self.spans)+1)
  s=dict(id=ident,d=d,tib=t,eng=e,**kw);self.spans.append(s)
  if e is not None:
   cur=0
   for i in range(n):
    pos=G.find_word(rows[self.n]['english'],e,cur,subword=kw.get('subword',False))
    assert pos>=0,(self.n,e,n)
    cur=pos+1
   self.positions[ident]=pos
  self.decisions.append({'id':ident,'tib':t,'eng':e,'english_occurrence':n if e else None,'decision':reason or 'Retain the local lexical correspondence; English supplied frame stays outside the span.'})
 def finish(self,note,omissions):
  path=base.parent/str(self.n);path.mkdir(exist_ok=True)
  order=sorted(self.positions,key=self.positions.get)
  seg=dict(seq=self.n,title=self.title,spans=self.spans,eng_order=order,note='PROVISIONAL machine alignment by Codex (English-first proposal). '+note)
  (path/'spec.json').write_text(json.dumps(dict(course='C05',segments=[seg]),ensure_ascii=False,indent=2)+'\n')
  (path/'errata.json').write_text('[]\n')
  (path/'decisions.json').write_text(json.dumps({'producer':'Codex','angle':'English-first','spans':self.decisions,'omissions':omissions,'intended_english_offsets':self.positions},ensure_ascii=False,indent=2)+'\n')
