def sha(b):return hashlib.sha256(b).hexdigest()
def write(name,obj):

def git(*args):return subprocess.check_output(['git','-C',str(ROOT),*args])
def preserve(original,relative,raw=None,**extra):
 raw=Path(original).read_bytes() if raw is None else raw
 p=HERE/relative;p.parent.mkdir(parents=True,exist_ok=True)
 if p.exists():assert p.read_bytes()==raw
 else:p.write_bytes(raw)
 record=dict(original=str(Path(original).absolute()),copy=relative,sha256=sha(raw),bytes=len(raw),**extra)
 if record not in manifest:manifest.append(record)
 assert p.read_bytes()==raw
 return record
supported={'.json','.md','.html','.stderr','.stdout','.log','.txt','.bin','.exit'}
def safe(p):return str(p) if p.suffix in supported else str(p)+('.txt' if p.suffix in {'.py','.sh','.sql'} else '.bin')
manifest=[];orig_files=[p for p in BATCH.rglob('*') if p.is_file() and HERE not in p.parents and '__pycache__' not in p.parts]

  dest=Path('reviewed-final')/Path(*rel.parts[1:])
 elif rel.parts[0] in {'english','tibetan'} and 'evidence' not in rel.parts and 'originals' not in rel.parts:
  dest=Path('proposal-inputs')/rel
 else:
  # Evidence snapshots can have identical bytes at many historical paths.
  raw=p.read_bytes();suffix=p.suffix if p.suffix in supported else p.suffix+'.txt' if p.suffix in {'.py','.sql','.sh'} else '.bin'
  dest=Path('proposal-inputs/evidence')/(sha(raw)+suffix)

   assert op.read_bytes()==raw
