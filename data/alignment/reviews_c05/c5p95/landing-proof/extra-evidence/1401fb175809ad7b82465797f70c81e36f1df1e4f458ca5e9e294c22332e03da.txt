def require(ok, detail):
    if not ok:
        raise ValueError(detail)


def pin(raw):
    return {'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}


def expected(row):
    return {k: row[k] for k in ('bytes', 'sha256')}


def read(path):
    p = Path(path)
    resolved = p.resolve(strict=True)
    require(resolved.is_file(), ('not regular file', str(p)))
    before = resolved.stat()
    raw = resolved.read_bytes()
    after = resolved.stat()
    attrs = ('st_dev', 'st_ino', 'st_size', 'st_mtime_ns', 'st_ctime_ns')
    require(all(getattr(before, k) == getattr(after, k) for k in attrs), ('unstable read', str(p)))
    require(len(raw) == after.st_size, ('short read', str(p)))
    return raw


def descriptor(path):
    p = Path(path)
    require(p.is_absolute(), ('relative original', str(p)))
    resolved = p.resolve(strict=True)
    # Preserve every symlink component, including directory aliases.
    aliases = []
    for q in (*reversed(p.parents), p):
        if q.is_symlink():
            aliases.append({'path': str(q), 'target': os.readlink(q)})
    return {'original': str(p), 'resolved': str(resolved), 'symlinks': aliases, **pin(read(p))}


def verified(row):
    require(descriptor(row['original']) == row, ('changed input identity', row['original']))
    raw = read(row['original'])
    require(pin(raw) == expected(row), ('changed input bytes', row['original']))
    return raw


def paths(root):
    root = Path(root)
    require(root.is_dir() and not root.is_symlink(), ('invalid root', str(root)))
    found = []
    for p in root.rglob('*'):
        require(not (p.is_symlink() and p.is_dir()), ('directory symlink enumeration unsupported', str(p)))
        require(not (p.is_symlink() and not p.exists()), ('broken symlink', str(p)))
        if p.is_file():
            found.append(str(p))
        else:
            require(p.is_dir(), ('unsupported filesystem entry', str(p)))
    return sorted(found)


def strict_json(raw):
    def no_constant(value):
        raise ValueError('nonstandard JSON constant: ' + value)
    return json.loads(raw.decode('utf-8'), parse_constant=no_constant)


def clean_json(path, raw):
    if Path(path).suffix != '.json' or len(raw) >= LIMIT:
        return False
    try:
        strict_json(raw)
    except (ValueError, UnicodeError):
        return False
    # Conservative literal predicate only; actual final Git diagnostic is mandatory.
    return (b'\r' not in raw and b'\t' not in raw and
            not any(line.endswith(b' ') for line in raw.split(b'\n')) and
            not raw.endswith(b'\n\n'))


def relative(value):
    p = Path(value)
    require(value and not p.is_absolute() and '..' not in p.parts, ('unsafe relative path', value))
    return p


def save(path, obj):
    raw = (json.dumps(obj, indent=2, ensure_ascii=True, allow_nan=False) + '\n').encode()
    with Path(path).open('xb') as f:
        f.write(raw)
    return pin(raw)
