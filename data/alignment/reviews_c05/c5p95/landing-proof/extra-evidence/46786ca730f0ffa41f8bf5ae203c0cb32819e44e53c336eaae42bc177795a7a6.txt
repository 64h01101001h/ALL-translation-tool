def dump(name, value):
    path=P/name

def pin(path):
    path=Path(path)
    b=path.read_bytes()
    return {'path':str(path),'bytes':len(b),'sha256':hashlib.sha256(b).hexdigest()}
copies=[]
def copy(path):
    path=Path(path)
    raw=path.read_bytes()
    h=hashlib.sha256(raw).hexdigest()
    dest=P/'original-evidence'/h
    dest.parent.mkdir(exist_ok=True)
    if not dest.exists(): dest.write_bytes(raw)
    assert dest.read_bytes()==raw
    entry={'original_path':str(path),'resolved_original_path':str(path.resolve()),'copy_path':str(dest),'bytes':len(raw),'sha256':h}
    copies.append(entry)
    return entry
