"""Reviewed later exact intake assembly. No bank/Git/source execution or self-banking."""
from pathlib import Path
import argparse
import hashlib
import importlib.machinery
import importlib.util
import json
import os
import sys
import traceback


def main():
    p = argparse.ArgumentParser(description=__doc__)
    for name in ('composer', 'composer-sha256', 'scope', 'scope-sha256', 'output'):
        p.add_argument('--' + name, required=True)
    p.add_argument('--execute-reviewed-supplement', action='store_true', required=True)
    args = p.parse_args()
    # Load only the exact reviewed inert helper module, never an evidence source.
    raw = Path(args.composer).read_bytes()
    raw_composer = raw
    if hashlib.sha256(raw).hexdigest() != args.composer_sha256:
        raise ValueError('reviewed composer identity changed')
    loader = importlib.machinery.SourceFileLoader('reviewed_final_composer', args.composer)
    spec = importlib.util.spec_from_loader(loader.name, loader)
    c = importlib.util.module_from_spec(spec)
    exec(compile(raw, args.composer, 'exec'), c.__dict__)
    c.require(os.environ.get('PYTHONDONTWRITEBYTECODE') == '1' and not sys.flags.optimize, 'bytecode/optimization contract')
    scope_raw = c.read(args.scope)
    c.require(c.pin(scope_raw)['sha256'] == args.scope_sha256, 'later scope pin')
    scope = c.strict_json(scope_raw)
    c.require(scope['schema'] == 'through282-final-supplement-v1' and scope['baseline'] == c.BASE, 'later scope schema')
    out = Path(args.output)
    c.require(out.is_absolute() and out.parent.resolve(strict=True) == c.I.resolve() and
              out.name == 'through282-final-bank-inventory-complete' and not out.exists(), 'fresh exact supplement output')
    c.require(scope['later_external_after_snapshot'], 'missing later caller boundary')
    out.mkdir(exist_ok=False)
    c.save(out / 'invocation.json', {'argv': sys.argv, 'cwd': os.getcwd(), 'scope_pin': c.pin(scope_raw),
        'environment_overrides': {'PYTHONDONTWRITEBYTECODE': '1', 'PYTHONOPTIMIZE': '0'}})
    coverage = []
    try:
        required_control_paths = {
            'initial_list': c.I / 'through282-final-bank-inventory/final-whitespace-inputs-initial.json',
            'initial_proof': c.I / 'through282-final-bank-inventory/proof.json',
            'initial_coverage': c.I / 'through282-final-bank-inventory/coverage-and-exclusion-map.json',
            'initial_scope': c.I / 'through282-final-intake-scope/final-source-scope.json',
            'ordinary_manifest': c.L / 'root-proof-copy-manifest-before-whitespace.json',
            'ordinary_extra_inputs': c.I / 'through282-ordinary-bank-inventory/extra-inputs.json'}
        c.require(set(scope['controls']) == set(required_control_paths), 'supplement control keys')
        for key, path in required_control_paths.items():
            c.require(scope['controls'][key]['original'] == str(path), ('supplement control path', key))
        controls = {k: c.verified(v) for k, v in scope['controls'].items()}
        first = c.strict_json(controls['initial_list'])
        proof = c.strict_json(controls['initial_proof'])
        initial_scope = c.strict_json(controls['initial_scope'])
        c.require(proof['status'] == 'PASS_INITIAL_COMPOSITION_REQUIRES_LATER_ROOT_SUPPLEMENT' and
                  proof['list_pin'] == c.pin(controls['initial_list']) and proof['scope_pin'] == c.pin(controls['initial_scope']) and
                  proof['coverage_pin'] == c.pin(controls['initial_coverage']), 'initial composition chain')
        c.require(scope['required_completed_roots'] == initial_scope['deferred_before_final_bank'], 'deferred scope must be exact')
        groups = scope['roots']
        c.require(len({g['root'] for g in groups}) == len(groups) and
                  set(scope['required_completed_roots']) <= {g['root'] for g in groups}, 'every deferred tree must now be complete')
        all_files = {}
        for group in groups:
            c.require(c.paths(group['root']) == [r['original'] for r in group['files']], ('later root enumeration', group['root']))
            for row in group['files']:
                c.require(not Path(row['resolved']).is_relative_to(out), 'supplement self reference')
                old_row = all_files.setdefault(row['original'], row)
                c.require(old_row == row, 'conflicting later alias identity')
        for row in scope['individual_files']:
            old_row = all_files.setdefault(row['original'], row)
            c.require(old_row == row, 'conflicting later individual identity')
        old = c.strict_json(controls['ordinary_manifest'])
        c.require(c.pin(controls['ordinary_manifest']) == c.expected(initial_scope['controls']['ordinary_manifest']) and
                  controls['ordinary_manifest'] == c.read(c.DEST / 'root-proof-copy-manifest-before-whitespace.json'), 'ordinary manifest unchanged')
        retained = {}
        for row in old['files']:
            raw = c.read(row['original'])
            c.require(c.pin(raw) == c.expected(row) and c.read(c.DEST / c.relative(row['copy'])) == raw, 'ordinary complete pair')
            retained.setdefault(str(Path(row['original']).resolve()), []).append(row)
        for row in old['external_actual_bytes_retained']:
            c.require(c.pin(c.read(row['path'])) == c.expected(row), 'ordinary external')
            retained.setdefault(str(Path(row['path']).resolve()), []).append(row)
        rows = list(first)
        by_source, targets = {}, {}
        for row in rows:
            source = str(Path(row['original']).resolve(strict=True))
            c.require(source == row['original'] and source not in by_source, 'duplicate/noncanonical initial resolved original')
            c.require(c.pin(c.read(source)) == c.expected(row), 'initial retained source changed')
            c.require(row['retention'] in ('bank', 'external'), 'initial retention schema')
            if row['retention'] == 'bank':
                c.require(row['copy'] not in targets and c.clean_json(source, c.read(source)), 'initial cleanJSON/target')
                c.relative(row['copy']); targets[row['copy']] = source
            else:
                c.require(row['disposition'], 'initial external disposition')
            by_source[source] = row
        for logical, item in sorted(all_files.items()):
            raw = c.verified(item); source = item['resolved']
            c.require(Path(source).is_relative_to(c.ART) and not Path(source).is_relative_to(c.DEST) and
                      not Path(source).is_relative_to(out), 'later source escape/self reference')
            if source in by_source:
                c.require(c.expected(by_source[source]) == c.pin(raw), 'duplicate original conflicting bytes')
                coverage.append({**item, 'disposition': 'already-in-initial-or-added-list', 'record': by_source[source]})
                continue
            if source in retained:
                c.require(all(c.expected(r) == c.pin(raw) for r in retained[source]), 'ordinary retained alias changed')
                coverage.append({**item, 'disposition': 'same-resolved-ordinary-original', 'records': retained[source]})
                continue
            if logical == source and not item['symlinks'] and c.clean_json(logical, raw):
                copy = 'final-safe-json/' + str(Path(logical).relative_to(c.ART))
                c.require(copy not in targets and not (c.DEST / c.relative(copy)).exists(), 'later copy collision')
                targets[copy] = source
                row = {'original': source, 'retention': 'bank', 'copy': copy, **c.pin(raw)}
            else:
                row = {'original': source, 'retention': 'external', **c.pin(raw),
                       'disposition': 'Complete original bytes retained externally at this exact absolute original path; no shortening, normalization or new bank alias'}
            rows.append(row); by_source[source] = row
            coverage.append({**item, 'disposition': row['retention'], 'record': row})
        for group in initial_scope['roots']:
            c.require(c.paths(group['root']) == [r['original'] for r in group['files']], 'initial source root changed')
            for item in group['files']: c.verified(item)
        for item in initial_scope['individual_files'] + initial_scope['late_batch_files']:
            c.verified(item)
        for item in initial_scope['controls'].values(): c.verified(item)
        c.require(rows[:len(first)] == first and len(by_source) == len(rows), 'initial rows altered or resolved originals duplicated')
        # Bind old extra LIST unchanged; only whitespace LIST is supplemented.
        c.require(c.pin(controls['ordinary_extra_inputs']) == c.expected(initial_scope['controls']['ordinary_extra_inputs']), 'old extra LIST altered')
        for group in groups:
            c.require(c.paths(group['root']) == [r['original'] for r in group['files']], 'later root changed during reads')
        for row in all_files.values(): c.verified(row)
        for row in scope['controls'].values(): c.verified(row)
        for row in rows: c.require(c.pin(c.read(row['original'])) == c.expected(row), 'retained source changed during reads')
        for row in old['files']:
            raw = c.read(row['original'])
            c.require(c.pin(raw) == c.expected(row) and c.read(c.DEST / c.relative(row['copy'])) == raw, 'ordinary pair changed')
        for row in old['external_actual_bytes_retained']:
            c.require(c.pin(c.read(row['path'])) == c.expected(row), 'ordinary external changed')
        c.require(c.read(args.scope) == scope_raw and Path(args.composer).read_bytes() == raw_composer, 'scope/helper changed')
        result = c.save(out / 'final-whitespace-inputs-complete.json', rows)
        c.save(out / 'coverage-and-exclusion-map.json', coverage)
        c.save(out / 'proof.json', {'status': 'PASS_COMPLETE_REVIEWED_SNAPSHOT_WITH_LATER_EXTERNAL_CALLER',
            'initial_rows': len(first), 'final_rows': len(rows), 'added_rows': len(rows)-len(first),
            'initial_rows_preserved_exactly': True, 'all_logical_aliases': len(all_files),
            'resolved_originals_unique': True, 'final_list': result, 'scope_pin': c.pin(scope_raw),
            'ordinary_extra_inputs_unchanged': scope['controls']['ordinary_extra_inputs'],
            'later_external_after_snapshot': scope['later_external_after_snapshot'],
            'actual_final_bank_and_diagnostic_equality_still_required': True, 'semantic_approval': False})
        print(json.dumps({'status': 'PASS_COMPLETE_REVIEWED_SNAPSHOT', 'final_list': result}))
    except Exception:
        (out / 'failure.txt').write_text(traceback.format_exc())
        c.save(out / 'partial-coverage.json', coverage)
        raise


if __name__ == '__main__':
    main()
