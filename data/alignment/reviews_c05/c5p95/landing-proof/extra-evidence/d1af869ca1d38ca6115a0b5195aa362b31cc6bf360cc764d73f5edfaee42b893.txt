Prepared while reconciliation runs. Execution requires the separate completed
semantic approval, root acceptance proof and exact already-landed page body.
No database, generated export or source text is modified by this helper.
"""

def review_segment_pin(row, name, review):
    assert name in ('spec', 'body') and type(row['seq']) is int
    flat = (name + '_bytes', name + '_sha256')
    if name in row:
        assert not any(k in row for k in flat), 'Ambiguous nested and flat pin fields'
        r = row[name]
        assert isinstance(r, dict) and set(r) == {'path', 'bytes', 'sha256'}
        expected = review / 'reviewed-final' / str(row['seq']) / ('spec.json' if name == 'spec' else 'body.html')
        assert Path(r['path']).is_absolute() and Path(r['path']) == expected
        result = {k: r[k] for k in ('bytes', 'sha256')}
    else:
        result = {'bytes': row[flat[0]], 'sha256': row[flat[1]]}
    assert type(result['bytes']) is int and result['bytes'] >= 0
    assert isinstance(result['sha256'], str) and len(result['sha256']) == 64
    assert all(c in '0123456789abcdef' for c in result['sha256'])
    return result

def pin(raw):
    return dict(bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest())

def git(*args):
    return subprocess.check_output(['git', *args], cwd=W, env=ENV)

def course_pages(raw):
    tree = ast.parse(raw)
    assignments = [n for n in tree.body if isinstance(n, ast.Assign)
                   and any(isinstance(t, ast.Name) and t.id == 'COURSES' for t in n.targets)]
    assert len(assignments) == 1
    root = assignments[0].value
    assert isinstance(root, ast.Dict)
    current = next(v for k, v in zip(root.keys, root.values)
                   if isinstance(k, ast.Constant) and k.value == 'C05')
    assert isinstance(current, ast.Dict)
    entries = next(v for k, v in zip(current.keys, current.values)
                   if isinstance(k, ast.Constant) and k.value == 'pages')
    return ast.literal_eval(entries)
