def require(ok, detail):
    if not ok:
        raise ValueError(detail)


def pin(raw):
    return {'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}


def capture(path):
    p = Path(path)
    resolved = p.resolve(strict=True)
    before = resolved.stat()
    raw = resolved.read_bytes()
    after = resolved.stat()
    attrs = ('st_dev', 'st_ino', 'st_size', 'st_mtime_ns', 'st_ctime_ns')
    require(resolved.is_file() and all(getattr(before, k) == getattr(after, k) for k in attrs), ('unstable input', str(p)))
    require(len(raw) == after.st_size, ('short read', str(p)))
    aliases = [{'path': str(q), 'target': os.readlink(q)} for q in (*reversed(p.parents), p) if q.is_symlink()]
    return raw, {'original': str(p), 'resolved': str(resolved), 'symlinks': aliases, **pin(raw)}


def write(name, raw):
    p = OUT / name
    require(p.is_relative_to(OUT), 'output escape')
    p.parent.mkdir(parents=True, exist_ok=True)
    with p.open('xb') as f:
        f.write(raw)
    return {'path': str(p), **pin(raw)}


def save(name, value):
    return write(name, (json.dumps(value, indent=2, ensure_ascii=True, allow_nan=False) + '\n').encode())
