"""Enumerate complete current evidence after accepted independent reproduction.

This script only writes a fresh external inventory. It never banks, stages,
changes sources, or manufactures acceptance. Later caller/ordinary-bank outputs
are explicitly outside its boundary and must be included by the final phase.
"""
from pathlib import Path
import datetime, hashlib, json, os

B = Path(__file__).resolve().parent
A = B.parent
I = A / 'integration'
W = A.parent / 'campaign-worktree'
O = I / 'through285-ordinary-bank-inventory'
BASE = 'fb9182dc6fca2230be3bc5169ba1288a22430280'
assert B.name == 'C05-283-285'
method = json.loads((B / 'root-ordinary-bank-method-release/proof.json').read_bytes())
assert method['status'] == 'PASS_ROOT_RELEASE_OF_REVIEWED_ORDINARY_METHOD'
assert method['composer']['path'] == str(Path(__file__).resolve())
assert hashlib.sha256(Path(__file__).read_bytes()).hexdigest() == method['composer']['sha256']
R = Path(method['reproduction_directory'])
ACCEPT = Path(method['reproduction_acceptance_directory'])

def pin(raw):
    return dict(bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest())

def read(path):
    before = path.stat()
    value = path.read_bytes()
    after = path.stat()
    fields = lambda s: (s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns, s.st_ctime_ns)
    assert fields(before) == fields(after), ('changed during read', str(path))
    return value

def j(path):
    return json.loads(read(path))

assert not O.exists()
accepted = j(ACCEPT / 'proof.json')
assert accepted['status'] == 'PASS' and accepted['head'] == BASE
assert j(B / 'baseline.json')['commit'] == BASE
assert j(B / 'root-payload-result-acceptance.json')['payload_files'] == 13
assert pin(read(ACCEPT / 'proof.json')) == method['required_reproduction_acceptance']
assert accepted['all_seven_reproduced_outputs_byte_exact'] and accepted['all_thirteen_payload_files_byte_exact'] and accepted['live_head_index_untracked_state_unchanged']
integration_inputs = [Path(p) for p in method['integration_inputs']]
assert len(integration_inputs) == len(set(integration_inputs)) and R in integration_inputs and ACCEPT in integration_inputs
assert I / 'through285-root-reconciliation-verification-attempt01' in integration_inputs
assert all(p.parent == I and p.name.startswith('through285') for p in integration_inputs)
roots = [B] + integration_inputs
assert all((p.is_dir() or p.is_file()) and not O.is_relative_to(p) for p in roots)
# The currently active outer recorder is an explicit later boundary. Source
# preparation, failed attempts, checkpoints, and every other B file are included.
deferred = [Path(p) for p in method['deferred_active_callers']]
assert len(deferred) == 1
assert all(p.parent in (B, I) and 'ordinary' in p.name and 'inventory' in p.name and p.name.endswith('caller') for p in deferred)
assert all(p.is_absolute() and not p.is_relative_to(B / 'semantic-review') for p in deferred)
expected_integration = {p for p in I.iterdir() if p.name.startswith('through285') and p != O and p not in deferred}
assert set(integration_inputs) == expected_integration, 'Every current through285 integration item must be explicitly included or the exact active caller deferred'
helpers = [A / name for name in [
    'record_external_step.py', 'land_reviewed_page_v2.py',
    'verify_semantic_acceptance.py', 'verify_review_copy_and_register.py',
    'run_reviewed_builders_once.py', 'verify_landed_batch.py',
    'verify_course_csv.py', 'run_generator_readonly.py',
    'run_tool_readonly_sqlite.py', 'copy_evidence_once_v2.py',
    'prepare_semantic_package_v2.py',
]]
assert all(p.is_file() for p in helpers)

def census():
    return {p for root in roots for p in ([root] if root.is_file() else root.rglob('*'))
            if p.is_file() and not any(p.is_relative_to(d) for d in deferred)} | set(helpers)

paths = census()
assert all(p.is_relative_to(A) for p in paths)
semantic = B / 'semantic-review'
landed = W / 'data/alignment/reviews_c05/c5p95'
sf = j(semantic / 'freeze.json')
assert pin(read(semantic / 'freeze.json'))['sha256'] == '8e398649b1152bd1604df1e5c3cc962c44742501484049fbe93c7015e23bf445'
assert {str(p.relative_to(semantic)) for p in semantic.rglob('*') if p.is_file()} == set(sf['files']) | {'freeze.json'}
bank_index = {}
for name in list(sf['files']) + ['freeze.json']:
    source, copy = semantic / name, landed / name
    raw = read(source)
    assert read(copy) == raw
    if name != 'freeze.json':
        assert pin(raw) == sf['files'][name]
    key = (len(raw), pin(raw)['sha256'])
    bank_index.setdefault(key, copy)

records, coverage, new_content = [], [], {}
supported = {'.json', '.md', '.html', '.stderr', '.stdout', '.log', '.txt', '.bin', '.exit'}
for source in sorted(paths):
    raw = read(source)
    value = pin(raw)
    key = (value['bytes'], value['sha256'])
    row = dict(original=str(source), resolved=str(source.resolve()),
               symlink=os.readlink(source) if source.is_symlink() else None, **value)
    if source.is_relative_to(B / 'landing'):
        # The banker itself inventories every landing file and validates it.
        row.update(coverage='automatic-complete-landing-inventory')
    elif key in bank_index:
        existing = bank_index[key]
        assert read(existing) == raw  # Hash equality alone never licenses reuse.
        disposition = 'Complete original retained here and byte-identical to the already banked semantic source at ' + str(existing)
        records.append(dict(original=str(source), retention='external', disposition=disposition, **value))
        row.update(coverage='explicit-existing-full-copy', banked_copy=str(existing))
    elif len(raw) >= 2_000_000:
        disposition = 'Complete large original retained at this exact pinned path; no truncation, normalization, or missing suffix selection.'
        records.append(dict(original=str(source), retention='external', disposition=disposition, **value))
        row.update(coverage='explicit-complete-external')
    else:
        if key in new_content:
            first, target = new_content[key]
            assert read(first) == raw
            # Downstream whitespace/staging maps require one bank record per
            # destination. Preserve this complete alias as an external record,
            # explicitly linking the first full copy planned by this inventory.
            disposition = 'Complete original retained here; byte-identical to ' + str(first) + ', whose full new bank copy is planned at landing-proof/' + target
            records.append(dict(original=str(source), retention='external', disposition=disposition, **value))
            row.update(coverage='explicit-deduplicated-new-copy', copy=target, first_original=str(first))
        else:
            suffix = source.suffix.lower()
            suffix = suffix if suffix in supported else '.txt' if suffix in {'.py', '.sh', '.diff', '.patch', '.jsonl'} else '.bin'
            target = 'extra-evidence/' + value['sha256'] + suffix
            new_content[key] = (source, target)
            records.append(dict(original=str(source), retention='bank', copy=target, **value))
            row.update(coverage='explicit-full-copy', copy=target)
    coverage.append(row)

assert paths == census(), 'Evidence tree changed during inventory; preserve attempt and retry only after resolving it'
for row in coverage:
    source = Path(row['original'])
    assert pin(read(source)) == {k: row[k] for k in ['bytes', 'sha256']}
    assert str(source.resolve()) == row['resolved']
    assert (os.readlink(source) if source.is_symlink() else None) == row['symlink']
# This one additional full copy preserves the existing generic record/alias.
# Its future automatic final-banker target must pre-exist the raw diagnostic.
route_source = I / 'through285-final-banking-stage-preparation/ordinary-required-final-banker-route.json'
route_raw = read(route_source)
assert pin(route_raw) == {'bytes': 1010, 'sha256': '72b5750f5977273d7dbf148c61249bc6cf750360334f3ca5b63265991622529e'}
required_route = json.loads(route_raw)['required_explicit_input']
assert required_route == dict(original=str(I / 'through285-final-banking-stage-preparation/final-candidates/bank_root_proofs_through285_final.py'), retention='bank', copy='root-helpers/bank_root_proofs_through285_final.py.txt', bytes=20978, sha256='2890f6e29393635e30a26e4ce5b849cfbd9a009f9b6d4f80ccc9b1cc9f86029f')
required_source = Path(required_route['original'])
assert required_source in paths and route_source in paths
required_raw = read(required_source)
assert pin(required_raw) == {k: required_route[k] for k in ('bytes', 'sha256')}
source_coverage = [row for row in coverage if row['original'] == str(required_source)]
assert len(source_coverage) == 1 and {k: source_coverage[0][k] for k in ('bytes', 'sha256')} == pin(required_raw)
assert str(required_source.resolve()) == source_coverage[0]['resolved']
assert (os.readlink(required_source) if required_source.is_symlink() else None) == source_coverage[0]['symlink']
generic_record_count = len(records)
generic_bank_targets = [row['copy'] for row in records if row['retention'] == 'bank']
assert len(generic_bank_targets) == len(set(generic_bank_targets)) == len(new_content)
assert required_route['copy'] not in generic_bank_targets
assert not (landed / 'landing-proof' / required_route['copy']).exists()
required_extra_copies = [dict(required_route)]
records.extend(required_extra_copies)
bank_targets = [row['copy'] for row in records if row['retention'] == 'bank']
assert len(bank_targets) == len(set(bank_targets)) == len(new_content) + len(required_extra_copies)
assert records[-1] == required_route and len(records) == generic_record_count + 1
assert read(required_source) == required_raw and read(route_source) == route_raw
assert paths == census(), 'Evidence tree changed during required extra-copy preparation'
O.mkdir()
for name, value in [
    ('extra-inputs.json', records), ('coverage-and-exclusion-map.json', coverage),
    ('scope.json', dict(roots=list(map(str, roots)), explicit_helper_files=list(map(str, helpers)),
                       deferred_active_caller_directories=list(map(str, deferred)),
                       later_outputs='This inventory and its later caller, ordinary bank output/caller, staging/whitespace and subsequent ledger/commit proofs must be captured at a truthful later boundary.')),
]:
    (O / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')
counts = {}
for row in coverage:
    counts[row['coverage']] = counts.get(row['coverage'], 0) + 1
proof = dict(status='PASS_COMPLETE_ACTUAL_PRE_ORDINARY_INVENTORY', utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),
             head=BASE, sources=len(coverage), counts=counts, records=len(records),
             distinct_new_full_copies=len(new_content) + len(required_extra_copies),
             generic_distinct_new_full_copies=len(new_content), generic_records_before_required_route=generic_record_count,
             required_additional_full_copies=len(required_extra_copies), required_extra_copy_records=required_extra_copies,
             required_extra_copy_source_coverage=source_coverage,
             required_extra_copy_proof=dict(path=str(route_source), **pin(route_raw)),
             manifest=pin(read(O / 'extra-inputs.json')),
             required_reproduction_acceptance=pin(read(ACCEPT / 'proof.json')),
             boundary='Every current B file and explicitly declared current through285 integration item plus eleven complete helper sources; only exact active caller directories are deferred. Full identical content may share a copy after complete-byte comparison. This is source inventory, not bank/stage/semantic/consumer acceptance.')
(O / 'proof.json').write_text(json.dumps(proof, indent=2) + '\n')
print(json.dumps(proof, indent=2))
