from pathlib import Path
import subprocess,sys,os,json,hashlib,datetime
D=Path(__file__).resolve().parent;O=D/'source-preparation-caller';O.mkdir()
def pin(b):return dict(bytes=len(b),sha256=hashlib.sha256(b).hexdigest())
argv=[sys.executable,'-B',str(D/'prepare-source.py')];env=dict(os.environ,PYTHONDONTWRITEBYTECODE='1',PYTHONOPTIMIZE='0')
c={'argv':argv,'cwd':str(D),'environment_overrides':{'PYTHONDONTWRITEBYTECODE':'1','PYTHONOPTIMIZE':'0'},'source':pin((D/'prepare-source.py').read_bytes())}
(O/'command.json').write_text(json.dumps(c,indent=2)+'\n');(O/'stdin.bin').write_bytes(b'')
t=datetime.datetime.now(datetime.timezone.utc).isoformat();r=subprocess.run(argv,cwd=D,env=env,input=b'',capture_output=True)
for n,b in [('stdout.bin',r.stdout),('stderr.bin',r.stderr),('exit.txt',(str(r.returncode)+'\n').encode())]:(O/n).write_bytes(b)
(O/'execution.json').write_text(json.dumps({**c,'started':t,'finished':datetime.datetime.now(datetime.timezone.utc).isoformat(),'exit':r.returncode,'stdout':pin(r.stdout),'stderr':pin(r.stderr)},indent=2)+'\n')
sys.stdout.buffer.write(r.stdout);sys.stderr.buffer.write(r.stderr);raise SystemExit(r.returncode)
