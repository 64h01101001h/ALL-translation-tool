def bp(raw):return {'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}
def raw(p):return Path(p).read_bytes()
def load(p):return json.loads(raw(p))
def ref(p):return {'path':str(p),**bp(raw(p))}
def check(p,pin):
 b=raw(p);assert bp(b)=={k:pin[k]for k in ['bytes','sha256']},str(p);assert raw(p)==b;return b
def relative(name):
 p=Path(name);assert isinstance(name,str)and name not in ('','.','..')and not p.is_absolute()and '..'not in p.parts
 return p
def check_freeze(d,fp,rows):
 expected={}
 for name,pin in rows:
  p=Path(name);name=str(p.relative_to(d))if p.is_absolute()else name;relative(name)
  assert name not in expected and name!=fp.name;assert (d/name).resolve().is_relative_to(d.resolve())
  check(d/name,pin);expected[name]=pin
 assert set(expected)=={str(p.relative_to(d))for p in d.rglob('*')if p.is_file()and p!=fp}
 return {'directory':str(d),'freeze':ref(fp),'members':len(expected)}
