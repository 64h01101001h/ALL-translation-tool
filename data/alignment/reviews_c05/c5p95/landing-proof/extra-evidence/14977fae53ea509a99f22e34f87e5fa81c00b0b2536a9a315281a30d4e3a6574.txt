def digest(raw): return hashlib.sha256(raw).hexdigest()
def pin(raw): return dict(bytes=len(raw), sha256=digest(raw))
def check_pin(raw, record):

    assert re.fullmatch('[0-9a-f]{64}', record['sha256'])
    assert pin(raw) == {k: record[k] for k in ('bytes', 'sha256')}
def relative(value):

def parse_log(raw):
    issues = defaultdict(lambda: defaultdict(list))
    header = re.compile(rb'^([^+\n]+):([0-9]+): (.+)$')
    for line in raw.split(b'\n'):
        match = header.fullmatch(line)
        if match:
            path, number, warning = match.groups()
            path = path.decode('utf-8'); relative(path)
            number = int(number); warning = warning.decode('ascii')
            assert number > 0 and warning in WARNINGS
            assert number not in issues[path][warning], 'Duplicate warning identity'
            issues[path][warning].append(number)
        else:
            assert not line.strip() or line.startswith(b'+'), ('Unknown diagnostic framing', line)
    return {path: dict(groups) for path, groups in issues.items()}

def literal_warnings(raw, grouped):
    """Common literal checks; EOF ownership is category-specific below."""
    lines = raw.split(b'\n')
    assert grouped and set(grouped) <= WARNINGS
    for kind, numbers in grouped.items():
        assert numbers and len(numbers) == len(set(numbers))
        for number in numbers:
            assert type(number) is int and 1 <= number <= len(lines)
            line = lines[number-1]
            if kind == 'trailing whitespace.': assert line.endswith((b' ', b'\t', b'\r'))
            elif kind == 'space before tab in indent.':
                indent = re.match(rb'^[ \t]*', line).group(); assert b' \t' in indent
            # EOF names/numbers are checked here; content belongs to its predicate.
    return lines

def diff_context_lines(raw):
    """Validate complete unified-diff framing/counts; return blank context lines.

No acceptance from merely finding ---/+++/@@ markers somewhere in payload.
    """
    lines = raw.split(b'\n'); lines = lines[:-1] if lines and lines[-1] == b'' else lines
    hunk = re.compile(rb'^@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@(?:.*)$')
    context = set(); old_left = new_left = 0; pair = False; pair_hunk = False; saw_hunk = False; i = 0
    while i < len(lines):
        line = lines[i]
        if old_left or new_left:
            assert line and line[:1] in (b' ', b'+', b'-')
            if line[:1] in (b' ', b'-'): old_left -= 1
            if line[:1] in (b' ', b'+'): new_left -= 1
            assert old_left >= 0 and new_left >= 0
            if line == b' ': context.add(i+1)
            i += 1
            if i < len(lines) and lines[i] == b'\\ No newline at end of file': i += 1
            continue
        if line.startswith(b'--- '):
            assert not pair or pair_hunk, 'File header pair has no complete hunk'
            assert len(line) > 4 and i+1 < len(lines) and lines[i+1].startswith(b'+++ ') and len(lines[i+1]) > 4
            pair = True; pair_hunk = False; i += 2; continue
        match = hunk.fullmatch(line)
        assert pair and match, ('Unrecognized or ambiguous diff framing', i+1, line)
        old_left = int(match.group(2) or b'1'); new_left = int(match.group(4) or b'1')
        assert old_left or new_left
        saw_hunk = True; pair_hunk = True; i += 1
    assert saw_hunk and pair_hunk and not old_left and not new_left
    return context

def ordinary_predicate(raw, grouped):
    lines = literal_warnings(raw, grouped)
    if set(grouped) == {'new blank line at EOF.'} and all(
            all(x == b'' for x in lines[n-1:]) for n in grouped['new blank line at EOF.']):
        return 'blank_eof'
    assert set(grouped) <= {'trailing whitespace.', 'new blank line at EOF.'}
    contexts = diff_context_lines(raw)
    for number in grouped.get('trailing whitespace.', []):
        assert number in contexts and lines[number-1] == b' '
    for number in grouped.get('new blank line at EOF.', []):
        # A terminal unified-diff context marker owns one ASCII space. Every
        # remaining content line must also be a parsed blank context marker.
        assert lines[-1] == b'' and number < len(lines)
        assert all(lines[i] == b' ' and i+1 in contexts for i in range(number-1, len(lines)-1))
    return 'diff_context'

def classify_physical(raw, grouped, physical):
    literal_warnings(raw, grouped)
    matches = []
    for item, witness in physical:
        if raw and raw in witness:
            start = 0
            while True:
                lo = witness.find(raw, start)
                if lo < 0: break
                matches.append(dict(witness=item, byte_range=[lo, lo+len(raw)])); start = lo+1
    if matches:
        if 'new blank line at EOF.' in grouped:
            # Preserve the original strict empty-LF rule. The added CRLF
            # exception exists only after whole-range ownership, never in
            # the report-line fallback; no fabricated all-blank CRLF fragment.
            lines = raw.split(b'\n')
            for number in grouped['new blank line at EOF.']:
                tail = lines[number-1:]
                if all(value == b'' for value in tail): continue
                assert raw.strip() and len(tail) >= 2 and tail[-1] == b''
                assert all(value == b'\r' for value in tail[:-1]), 'Not exact empty CRLF source tail'
        return dict(kind='physical_bytes', full_range_witnesses=matches)
    assert set(grouped) == {'trailing whitespace.'}
    lines = raw.split(b'\n'); witnesses = {}
    for number in grouped['trailing whitespace.']:
        value = lines[number-1]; assert value and value.strip(), 'Empty physical-line promotion refused'
        matches = []
        for item, source in physical:
            for match in re.finditer(re.escape(value), source):
                lo, hi = match.span()
                if (lo == 0 or source[lo-1:lo] == b'\n') and source[hi:hi+1] in (b'\r', b'\n'):
                    matches.append(dict(witness=item, byte_range=[lo, hi]))
        assert matches, ('No complete original physical line', number)
        witnesses[str(number)] = matches
    return dict(kind='physical_source_lines', line_witnesses=witnesses)

    def capture_root(self, supplied, role):
        alias = Path(supplied).absolute(); target = alias.resolve(strict=True)
        stat = target.stat()
        self.root_aliases.append(dict(role=role, supplied_path=str(supplied), input_path=str(alias),

        return target

        for record in self.root_aliases:
            target = Path(record['input_path']).resolve(strict=True)
            assert target == Path(record['path']), ('Root alias retargeted', record['role'])
            stat = target.stat()
            assert (stat.st_dev, stat.st_ino) == (record['device'], record['inode']), ('Root identity changed', record['role'])

        for rec in root['files']:
            rel = str(relative(rec['copy'])); assert rel not in root_rows; root_rows[rel] = rec
        log = self.read(self.a.raw_log); stderr = self.read(self.a.raw_stderr); code = int(self.read(self.a.raw_exit)); command = self.obj(self.a.raw_command)
        assert command['argv'] == ['git','diff','--cached','--check'] and Path(command['cwd']).resolve() == self.w
        assert not stderr and code in (0,2)
        issues = parse_log(log); assert bool(issues) == (code == 2)
        if self.a.mode == 'audit':
            actual, actual_code = self.git('diff','--cached','--check',allowed=(0,2)); assert actual == log and actual_code == code
            assert self.a.prepared
            prepared = self.obj(Path(self.a.prepared)/'prepared-pins.json')
            assert prepared['head'] == HEAD and prepared['raw_log'] == pin(log) and prepared['review_freeze'] == pin(freeze_raw) and prepared['root_manifest'] == pin(root_raw)
        else: assert not self.a.prepared
        build = self.obj(self.a.build_proof); assert build['status'] == 'PASS' and build['baseline_commit'] == HEAD and build['page'] == 94 and build['first_segment'] == 280
        additional, additional_values = load_additional_root(self.read, self.read(HERE/'additional-root-descriptors.json'), root_raw, log)
        additional_paths_used = set()
        for path, grouped in issues.items():
            raw = self.staged(path); lines = literal_warnings(raw,grouped); witness = {}
            if path in CSV:
                assert digest(raw) == build['output_sha256'][path]
                assert set(grouped) == {'trailing whitespace.'} and all(lines[n-1].endswith(b'\r') for n in grouped['trailing whitespace.'])
                witness = dict(kind='canonical_csv_crlf',build_proof=str(Path(self.a.build_proof).resolve()))
            else:
                assert path.startswith(PREFIX); rel = path[len(PREFIX):]
                if rel.startswith('landing-proof/'):
                    key = rel[len('landing-proof/'):]; assert key in root_rows
                    rec = root_rows[key]; check_pin(raw,rec); original = self.read(rec['original']); assert original == raw
                    try: kind = ordinary_predicate(raw,grouped)
                    except AssertionError:
                        witness = additional_root_predicate(raw,grouped,path,rec,additional,additional_values,build,physical)
                        witness['root_original_record'] = rec
                        additional_paths_used.add(path)
                    else: witness = dict(kind=kind,root_original_record=rec)
                else:
                    assert raw == frozen_read(rel)
                    try: kind = ordinary_predicate(raw,grouped)
                    except AssertionError:
                        witness = classify_physical(raw,grouped,physical)
                    else: witness = dict(kind=kind)
            record = dict(**pin(raw),issues=grouped,witness=witness)
            if self.a.mode == 'audit': assert record == prepared['files'][path]
            self.checks[path] = record; self.save('partial-classification.json',self.checks)
        assert additional_paths_used == {entry['path'] for entry in additional['entries']}
        if self.a.mode == 'audit': assert set(self.checks) == set(prepared['files'])
        assert self.git('ls-files','--stage','-z')[0] == index
        assert self.git('rev-parse','HEAD')[0].decode().strip() == head
        for record in self.inputs:
            alias = Path(record['input_path']); assert alias.resolve(strict=True) == Path(record['path'])
            original = alias.read_bytes(); check_pin(original,record); assert original == (self.out/record['copy']).read_bytes()
        self.recheck_roots()
        metadata = dict(head=HEAD,raw_log=pin(log),review_freeze=pin(freeze_raw),root_manifest=pin(root_raw),files=self.checks)
        self.save('prepared-pins.json',metadata)
        counts = Counter()
        for record in self.checks.values(): counts[record['witness']['kind']] += sum(map(len,record['issues'].values()))
        proof = dict(checked_at=datetime.datetime.now(datetime.timezone.utc).isoformat(),mode=self.a.mode,status='PASS classification; independent tooling review required' if self.a.mode=='prepare' else 'PASS actual staged diagnostic reproduced and predicates verified',baseline_head=HEAD,review_freeze_sha256=digest(freeze_raw),git_diff_check_exit=code,warning_count=sum(counts.values()),all_warnings_accounted_for=True,raw_log_bytes=len(log),raw_log_sha256=digest(log),raw_log_kept_external=str(Path(self.a.raw_log).resolve()),classes=dict(counts),files={p:dict(bytes=x['bytes'],sha256=x['sha256'],issues=x['issues'],reason=x['witness']['kind']) for p,x in self.checks.items()},semantic_approval=False,limitations='Literal whitespace provenance only. Frozen bytes alone never excuse warnings. Static inventory was not used as actual Git diagnostics. Full freeze/semantic approval, bank retention and staging boundaries remain separate gates.')
        self.save('whitespace-classification-preflight.json' if self.a.mode=='prepare' else 'source-whitespace-proof.json',proof)
        return {k:v for k,v in proof.items() if k!='files'}

def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('mode',choices=['prepare','audit'])
    for key in ('worktree','review','root-manifest','root-manifest-bank-path','physical-descriptors','raw-log','raw-stderr','raw-exit','raw-command','build-proof','output'): parser.add_argument('--'+key,required=True)
    parser.add_argument('--prepared')
    args = parser.parse_args()
    assert __debug__, 'Optimized Python forbidden'
    require_runtime_release(args)
    run = Run(args)
    try: result = run.run()
    except BaseException:
        run.save('failure.json',dict(utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),traceback=traceback.format_exc(),partial_files=list(run.checks),disposition='Keep this entire fresh attempt unchanged, including all copied full inputs, commands, raw diagnostics and partial classification; use a distinct output for any retry.'))
        raise
    print(json.dumps(result,indent=2))
if __name__ == '__main__': main()
