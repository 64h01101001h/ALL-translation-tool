def pin(path):
    raw=path.read_bytes()
    return {'path':str(path),'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}
copies=[]
def copy(path):
    raw=path.read_bytes()
    h=hashlib.sha256(raw).hexdigest()
    dest=P/'original-evidence'/h
    if not dest.exists():dest.write_bytes(raw)
    assert dest.read_bytes()==raw
    copies.append({'original_path':str(path),'resolved_original_path':str(path.resolve()),'copy_path':str(dest),'bytes':len(raw),'sha256':h})
    if path.resolve()!=path:
        actual=path.resolve()
        assert actual.read_bytes()==raw
        copies.append({'original_path':str(actual),'resolved_original_path':str(actual),'copy_path':str(dest),'bytes':len(raw),'sha256':h})
    return copies[-1]
