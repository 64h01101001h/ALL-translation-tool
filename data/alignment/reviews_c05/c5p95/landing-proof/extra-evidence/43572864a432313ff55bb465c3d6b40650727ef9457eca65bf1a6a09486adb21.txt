"""Capture the actual source-preparation transport; never invokes the inventory composer."""
from pathlib import Path
import datetime
import hashlib
import json
import os
import subprocess
import sys

ROOT = Path('/Users/adamderickandrade/Documents/ChatGPT/Geshe Michael Roach Tib _ Eng Alignment')
OUT = ROOT / 'campaign-artifacts/integration/through282-final-bank-inventory-preparation'
CALLER = OUT / 'code-only-caller'
CALLER.mkdir(exist_ok=False)

def pin(raw):
    return {'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}

def save(name, value):
    raw = (json.dumps(value, indent=2, ensure_ascii=True, allow_nan=False) + '\n').encode()
    with (CALLER / name).open('xb') as f:
        f.write(raw)

argv = [sys.executable, '-B', str(OUT / 'prepare-source.py.txt')]
env = os.environ.copy()
env.update(PYTHONDONTWRITEBYTECODE='1', PYTHONOPTIMIZE='0')
start = datetime.datetime.now(datetime.timezone.utc).isoformat()
save('invocation.json', {'kind': 'ACTUAL_SOURCE_PREPARATION_CALLER', 'argv': argv, 'cwd': str(ROOT),
                        'environment_overrides': {'PYTHONDONTWRITEBYTECODE': '1', 'PYTHONOPTIMIZE': '0'},
                        'started_utc': start, 'stdin': {'bytes': 0, 'sha256': hashlib.sha256(b'').hexdigest()}})
run = subprocess.run(argv, cwd=ROOT, env=env, input=b'', stdout=subprocess.PIPE, stderr=subprocess.PIPE, check=False)
end = datetime.datetime.now(datetime.timezone.utc).isoformat()
for name, raw in [('stdin.bin', b''), ('stdout.txt', run.stdout), ('stderr.txt', run.stderr), ('exit.txt', (str(run.returncode) + '\n').encode())]:
    with (CALLER / name).open('xb') as f:
        f.write(raw)
result = {'kind': 'ACTUAL_RAW_SOURCE_PREPARATION_TRANSPORT_NOT_COMPOSER_EXECUTION',
          'started_utc': start, 'finished_utc': end, 'exit': run.returncode,
          'stdout': pin(run.stdout), 'stderr': pin(run.stderr), 'target_executed': 'prepare-source.py.txt',
          'composer_imports_or_invocations': 0}
save('result.json', result)
print(json.dumps(result, sort_keys=True))
raise SystemExit(run.returncode)
