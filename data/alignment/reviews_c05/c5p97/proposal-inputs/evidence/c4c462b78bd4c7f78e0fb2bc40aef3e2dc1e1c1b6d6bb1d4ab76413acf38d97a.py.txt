import json,sqlite3,gzip

con=sqlite3.connect((W/'build/hgm_spine_v27_2.db').as_uri()+'?mode=ro');con.row_factory=sqlite3.Row
es=json.loads((E/'lexical-entries.json').read_text());qs=json.loads((E/'lexical-queries.json').read_text());ms=json.loads((E/'master-records.json').read_text());mq=json.loads((E/'master-queries.json').read_text())
