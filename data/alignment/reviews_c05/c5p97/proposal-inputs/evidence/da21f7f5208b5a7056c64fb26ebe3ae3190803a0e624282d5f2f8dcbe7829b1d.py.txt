def make(seq,title,items,note,omissions,discussion):
 out=E/str(seq);out.mkdir(exist_ok=True);spans=[];positions=[];decisions=[];tcur=0;parent=None
 src=sources[seq]
 for item in items:
  sid,d,tib,eng,n,term,why,*extra=item
  sp={'id':sid,'d':d,'tib':tib,'eng':eng}
  if eng is None:sp['nul']=why
  if extra:sp.update(extra[0])
  if d==7:
   assert parent is not None
   ta=src['wylie'].find(tib,parent[0],parent[1]);assert ta>=0
  else:
   ta=src['wylie'].find(tib,tcur);assert ta>=0,(seq,sid,tib)
   tcur=ta+len(tib)
   if d==5:parent=(ta,tcur)
  ea=None
  if eng is not None:
   hits=[m.start() for m in re.finditer(re.escape(eng),src['english']) if (not eng[0].isalnum() or m.start()==0 or not src['english'][m.start()-1].isalnum()) and (not eng[-1].isalnum() or m.end()==len(src['english']) or not src['english'][m.end()].isalnum())]
   assert len(hits)>=n,(seq,sid,eng,hits,n);ea=hits[n-1]
   if d!=7:positions.append((ea,sid))
  spans.append(sp)
  decisions.append({'span_id':sid,'depth':d,'tibetan':tib,'english':eng,'intended_tibetan_range_0based_halfopen':[ta,ta+len(tib)],'intended_english_occurrence_1based':n if eng is not None else None,'intended_english_range_0based_halfopen':[ea,ea+len(eng)] if ea is not None else None,'source_row':'evidence/source-context-full-rows.json / rows where course C05 and seq '+str(seq),'hgm_query_ref':{'file':'evidence/hgm-lexical.json','query_index_0based':lex_terms[term],'query_wylie':term} if term in lex_terms else None,'judgment':why})
 spec={'course':'C05','segments':[{'seq':seq,'title':title,'spans':spans,'eng_order':[sid for _,sid in sorted(positions)],'note':'PROVISIONAL machine alignment by Codex, independent English-first proposal. '+note+' No hgm_gloss change; mechanical fidelity does not establish semantic approval.'}]}
 (out/'spec.json').write_text(json.dumps(spec,ensure_ascii=False,indent=2)+'\n');(out/'errata.json').write_text('[]\n')
 (out/'decisions.json').write_text(json.dumps({'actual_producer':'Codex','angle':'English-first','governing_commit':'129e86afae8431aacfb1ee1cee65d8635dd265b8','source_baseline_boundary':285,'source_role':'Original proposal source baseline only; does not claim through288 landing.','full_local_source_argument_read':'C05:283–297; target and both neighboring argument continuations retained verbatim in evidence/source-context-full-rows.json.','spans':decisions,'omissions':omissions,'discussion':discussion,'errata_conclusion':{'new_proposals':0,'screen':'evidence/errata-screen.json','source_preserved_verbatim':True,'physical_identity':'evidence/physical-source.json','corpus_queries':['exact_text_'+str(seq)],'limits':'Textual matches are not independent publication or ingestion-lineage proof; C15:436 differs in English punctuation. No defect inferred merely from translation expansion or differing punctuation.'}},ensure_ascii=False,indent=2)+'\n')
