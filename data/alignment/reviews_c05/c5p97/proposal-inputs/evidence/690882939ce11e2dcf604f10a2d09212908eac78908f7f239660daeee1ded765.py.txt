class SourceParser(html.parser.HTMLParser):
    def __init__(self):super().__init__(convert_charrefs=True);self.side=None;self.depth=0;self.text={'tib':[],'eng':[]};self.null_depth=None
    def handle_starttag(self,tag,attrs):
        attrs=dict(attrs)
        if tag=='div' and attrs.get('class') in ['tib','eng']:
            self.side=attrs['class'];self.depth=1
        elif self.side and tag=='div': self.depth+=1
        if self.side and tag=='span' and 'nul' in attrs.get('class','').split(): self.null_depth=1
        elif self.null_depth is not None and tag=='span':self.null_depth+=1
    def handle_endtag(self,tag):
        if self.null_depth is not None and tag=='span':
            self.null_depth-=1
            if self.null_depth==0:self.null_depth=None
        if self.side and tag=='div':
            self.depth-=1
            if self.depth==0:self.side=None
    def handle_data(self,data):
        if self.side and self.null_depth is None:self.text[self.side].append(data)
