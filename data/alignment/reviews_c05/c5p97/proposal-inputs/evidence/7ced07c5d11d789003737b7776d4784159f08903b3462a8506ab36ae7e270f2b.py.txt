from pathlib import Path
import json,re
A=Path(__file__).resolve().parent; B=A.parent
sources={r['seq']:r for r in json.loads((B/'source.json').read_text())}

def make(seq,title,items,notes,omissions,decisions):
    src=sources[seq]; spans=[]; evidence=[]; cur=0
    for n,item in enumerate(items,1):
        d,tib,eng,occ,*extra=item
        sid=('p' if d==6 else 'x' if d==3 else 'w')+str(n)
        sp={'id':sid,'d':d,'tib':tib,'eng':eng}
        if eng is None: sp.update({'cls':'grammar','nul':extra[0]})
        if extra and isinstance(extra[-1],dict): sp.update(extra[-1])
        # This authoring inventory records explicit intended occurrences, independently
        # checked against actual canonical ranges after generation.
        hits=[m.start() for m in re.finditer(re.escape(tib),src['wylie'])]
        thits=[i for i in hits if i>=cur and (i==0 or src['wylie'][i-1] in ' ,')]
        assert thits,(seq,tib)
        ti=thits[0]; cur=ti+len(tib)
        ei=None
        if eng is not None:
            ehits=[m.start() for m in re.finditer(re.escape(eng),src['english']) if (m.start()==0 or not src['english'][m.start()-1].isalnum()) and (m.end()==len(src['english']) or not src['english'][m.end()].isalnum())]
            assert len(ehits)>=occ,(seq,eng,occ,ehits)
            ei=ehits[occ-1]
        spans.append(sp)
        evidence.append({'id':sid,'d':d,'tib':tib,'eng':eng,'intended_tib_range':[ti,ti+len(tib)],'intended_tib_exact_occurrence':hits.index(ti)+1,'intended_eng_range':None if ei is None else [ei,ei+len(eng)],'intended_eng_word_occurrence':occ if eng is not None else None,'role_note':extra[0] if extra and isinstance(extra[0],str) else None})
    order=[r['id'] for r in sorted([r for r in evidence if r['eng'] is not None],key=lambda r:r['intended_eng_range'][0])]
    spec={'course':'C05','segments':[{'seq':seq,'title':title,'spans':spans,'eng_order':order,'note':'PROVISIONAL machine alignment by Codex (GPT-6 Astra), Tibetan-first independent proposal. GMR English and original Wylie remain verbatim. No hgm_gloss changes. '+notes}]}
    D=A/str(seq)
    (D/'spec.json').write_text(json.dumps(spec,ensure_ascii=False,indent=2)+'\n')
    (D/'errata.json').write_text('[]\n')
    (D/'decisions.json').write_text(json.dumps({'seq':seq,'source_spine_id':src['id'],'source_evidence':'../evidence/corpus-rows.json','entries':evidence,'omissions':omissions,'contentious_decisions':decisions},ensure_ascii=False,indent=2)+'\n')
