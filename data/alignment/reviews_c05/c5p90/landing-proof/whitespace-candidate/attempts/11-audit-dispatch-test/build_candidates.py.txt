import difflib

HERE=Path(__file__).resolve().parent

def save(name,old,new):
    assert new!=old
    (HERE/name).write_text(new)
    (HERE/(name+'.diff')).write_text(''.join(difflib.unified_diff(old.splitlines(True),new.splitlines(True),fromfile='baseline/'+name,tofile=name)))
