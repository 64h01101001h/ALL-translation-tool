import hashlib,json,os,subprocess,sys,time
from pathlib import Path
p=Path(__file__).resolve().parent
out=p/'attempts'/sys.argv[1];out.mkdir(parents=True,exist_ok=False)
pins={}
for source in p.glob('*.py'):
    raw=source.read_bytes();(out/source.name).write_bytes(raw)
    pins[source.name]={'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}
argv=sys.argv[2:];env=dict(os.environ,PYTHONDONTWRITEBYTECODE='1')
(out/'command.json').write_text(json.dumps(dict(argv=argv,cwd=str(p),stdin_bytes=0,environment_overrides={k:env[k] for k in ['PYTHONDONTWRITEBYTECODE','WITNESS_MODULE'] if k in env},code_pins=pins),indent=2)+'\n')
(out/'stdin.bin').write_bytes(b'');start=time.perf_counter()
r=subprocess.run(argv,cwd=p,env=env,input=b'',capture_output=True)
(out/'stdout.bin').write_bytes(r.stdout);(out/'stderr.bin').write_bytes(r.stderr);(out/'exit.txt').write_text(str(r.returncode)+'\n')
(out/'result.json').write_text(json.dumps({'exit':r.returncode,'wall_seconds':time.perf_counter()-start},indent=2)+'\n')
sys.stdout.buffer.write(r.stdout);sys.stderr.buffer.write(r.stderr);sys.exit(r.returncode)
