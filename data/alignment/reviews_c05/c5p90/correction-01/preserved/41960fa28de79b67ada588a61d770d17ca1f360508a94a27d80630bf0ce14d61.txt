"""Mechanically verify the actual correction before distinct scoped review."""
from pathlib import Path
from collections import Counter
import datetime, difflib, hashlib, json, os, sqlite3, subprocess, sys

B = Path(__file__).resolve().parent
C = B / 'reconciliation-correction-01'
H = B / 'pre-correction-01'
R = B / 'reconciled/268'
W = B.parent.parent / 'campaign-worktree'
O = B / 'root-correction-01-verification'
assert not O.exists(), 'Preserve prior attempts'
O.mkdir()

def read(p): return json.loads(p.read_text())
def pin(p):
    raw = p.read_bytes()
    return {'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}
def check(p, expected):
    assert pin(p) == {k: expected[k] for k in ('bytes', 'sha256')}, p
def write(p, value): p.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

freezes = []
for directory in [C, R, B/'reconciled/269', B/'reconciled/270']:
    f = read(directory/'freeze.json')
    expected = {r['relative_path']: r for r in f['files']}
    actual = {str(p.relative_to(directory)) for p in directory.rglob('*') if p.is_file() and p != directory/'freeze.json'}
    assert actual == set(expected) and len(actual) == f['file_count']
    for rel, row in expected.items(): check(directory/rel, row)
    freezes.append({'path': str(directory/'freeze.json'), **pin(directory/'freeze.json'), 'files': len(actual)})
for row in read(B/'root-initial-semantic-verification.json')['history_files']:
    check(Path(row['copy']), row)
for p in (H/'semantic-review').rglob('*'):
    if p.is_file(): assert p.read_bytes() == (B/'semantic-review'/p.relative_to(H/'semantic-review')).read_bytes()

before = read(C/'before-files.json')
after = read(C/'after-files.json')
actual = {str(p.relative_to(B)): {'path': str(p), **pin(p)} for seq in [268,269,270] for p in (B/f'reconciled/{seq}').rglob('*') if p.is_file()}
assert actual == after
assert set(before) == {str(p.relative_to(H)) for p in (H/'reconciled').rglob('*') if p.is_file()}
for rel, row in before.items(): check(H/rel, row)
changes = []
diff = []
for rel in sorted(set(before)|set(after)):
    a, b = before.get(rel), after.get(rel)
    if a == b: continue
    assert rel.startswith('reconciled/268/') and b is not None
    changes.append({'path':str(B/rel), 'before':a, 'after':b, 'immutable_before_path':str(H/rel) if a else None, 'status':'added' if not a else 'changed'})
    old = (H/rel).read_bytes() if a else b''
    new = (B/rel).read_bytes()
    try:
        diff.extend(difflib.unified_diff(old.decode('utf-8').splitlines(True),new.decode('utf-8').splitlines(True),fromfile=str(H/rel) if a else '/dev/null',tofile=str(B/rel),n=5))
    except UnicodeDecodeError:
        diff.append('Binary exact-byte change; see full before/after pins and retained files: '+rel+'\n')
assert changes == read(C/'before-after-manifest.json')['changes']
assert Counter(x['status'] for x in changes) == {'changed':13,'added':39}
assert ''.join(diff).encode() == (C/'exact-correction.diff.txt').read_bytes()
expected_changed = {'spec.json','body.html','errata.json','errata-analysis.json','report.md','generator-proof.json','batch-counts.json','original-dispositions.json','omissions.json','original-to-final.diff.txt','final-audit.json','supported-copy-preflight.json','freeze.json'}
assert {Path(x['path']).name for x in changes if x['status']=='changed'} == expected_changed

tuples = []
originals = 0
for seq in [268,269,270]:
    d = B/f'reconciled/{seq}'
    h = H/f'reconciled/{seq}'
    old, new = read(h/'spec.json'), read(d/'spec.json')
    if seq == 268:
        assert old['segments'][0]['note'] != new['segments'][0]['note']
        old['segments'][0]['note'] = new['segments'][0]['note']
    assert old == new
    assert [list(s) for s in old['segments'][0]['spans']] == [list(s) for s in new['segments'][0]['spans']]
    assert read(d/'errata.json') == []
    current_tuples = read(d/'resolved-spans.json')
    assert current_tuples == read(h/'resolved-spans.json')
    tuples.extend(current_tuples)
    a, b = read(h/'original-dispositions.json'), read(d/'original-dispositions.json')
    assert len(a) == len(b)
    for previous, current in zip(a,b):
        if previous != current:
            assert seq == 268 and previous['original_identity']['angle']=='tibetan' and previous['original_identity']['id']=='w17'
            assert {k:v for k,v in previous.items() if k!='reason'} == {k:v for k,v in current.items() if k!='reason'}
    originals += len(b)
    for name in ['final-decisions.json','unwrapped-extents.json','proposed-span-head-allowances.json']:
        assert (d/name).read_bytes() == (h/name).read_bytes()
assert tuples == read(B/'root-reconciliation-verification/resolved-spans.json')
assert len(tuples)==123 and originals==227

assert len(candidate['five_grounds']) == 5
