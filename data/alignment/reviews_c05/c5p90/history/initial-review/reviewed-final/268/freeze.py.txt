import pathlib, json, hashlib, datetime, os, sys

R = pathlib.Path(__file__).resolve().parent
def pin(p):
    b=p.read_bytes()
    return {'path':str(p),'sha256':hashlib.sha256(b).hexdigest(),'bytes':len(b)}
def put(p,x): p.write_text(json.dumps(x,ensure_ascii=False,indent=2)+'\n')

assert sys.dont_write_bytecode and os.environ.get('PYTHONDONTWRITEBYTECODE')=='1'
assert all(not (R.parent/str(seq)/'freeze.json').exists() for seq in (268,269,270))
assert json.loads((R/'final-audit.json').read_text())['status']=='PASS'
assert (R/'finalize-attempt01.stderr.txt').read_bytes()==b''
assert json.loads((R/'finalize-attempt01.stdout.txt').read_text())['status']=='PASS'
(R/'finalize-attempt01.exit.txt').write_text('0\n')
(R/'freeze.py.txt').write_bytes(pathlib.Path(__file__).read_bytes())
put(R/'freeze-stage-execution.json',{
    'producer':'reused Codex non-author c05_253_255_reconcile',
    'pre_freeze_checks':'PASS; finalizer actual tool exit0 and empty stderr were observed before this invocation.',
    'finalizer':{
        'argv':['python3','-B','campaign-artifacts/C05-268-270/reconciled/268/finalize.py'],
        'cwd':str(R.parents[3]),'environment':{'PYTHONDONTWRITEBYTECODE':'1'},
        'exit':0,'stdout':pin(R/'finalize-attempt01.stdout.txt'),'stderr':pin(R/'finalize-attempt01.stderr.txt'),
        'script':pin(R/'finalize.py')},
    'freeze_argv':['python3','-B',*sys.argv],
    'freeze_cwd':os.getcwd(),'environment':{'PYTHONDONTWRITEBYTECODE':'1'},
    'script':pin(pathlib.Path(__file__).resolve()),
    'supported_copy':pin(R/'freeze.py.txt'),
    'note':'This record precedes the freeze. The final tool return establishes process exit; no post-freeze file mutation is performed.'})
put(R/'freeze-packaging-addendum.json',{
    'additional_source_to_copy_mappings':[{'original':str(pathlib.Path(__file__).resolve()),'copy':'freeze.py.txt','bytes':pin(R/'freeze.py.txt')['bytes'],'sha256':pin(R/'freeze.py.txt')['sha256']}],
    'copy_equals_original':(R/'freeze.py.txt').read_bytes()==pathlib.Path(__file__).read_bytes(),
    'supported_suffix':'.txt','prior_complete_preflight':'supported-copy-preflight.json',
    'combined_material_mappings':221,
    'scope':'Only current reconciled/268–270. Complete freeze retains raw .py and .png plus their supported byte-identical copies.'})
results=[]
for seq in (269,270,268):
    D=R.parent/str(seq)
    files=[{**pin(p),'relative_path':str(p.relative_to(D))} for p in sorted(D.rglob('*')) if p.is_file() and p!=D/'freeze.json']
    frozen={'seq':seq,'status':'PROVISIONAL frozen reconciliation','producer':'reused Codex non-author c05_253_255_reconcile',
            'frozen_at_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),
            'root':str(D),'self_exclusion':'Only this freeze.json is excluded from its own enumeration.',
            'file_count':len(files),'total_bytes':sum(x['bytes'] for x in files),'files':files}
    put(D/'freeze.json',frozen)
    actual=sorted(str(p.relative_to(D)) for p in D.rglob('*') if p.is_file() and p!=D/'freeze.json')
    assert actual==[x['relative_path'] for x in files]
    for item in files: assert pin(pathlib.Path(item['path']))=={k:item[k] for k in ('path','sha256','bytes')}
    results.append({'seq':seq,'file_count':len(files),'freeze':pin(D/'freeze.json')})
print(json.dumps({'status':'PASS','freeze_complete':results}))
