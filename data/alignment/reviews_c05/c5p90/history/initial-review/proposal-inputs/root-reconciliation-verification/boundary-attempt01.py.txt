"""Mechanical root handoff check. Root authored Tibetan; no semantic approval."""
from pathlib import Path
from collections import Counter
import datetime, hashlib, json, sqlite3, subprocess

b = Path(__file__).resolve().parent
wt = b.parent.parent / 'campaign-worktree'
out = b / 'root-reconciliation-verification'
target = out / 'root-frozen-evidence-check.json'
assert not target.exists(), 'Preserve any earlier boundary check before retry.'
key7 = ('seq','id','d','tib','eng','tib_range','eng_range')
key8 = ('angle',) + key7
def read(p): return json.loads(p.read_bytes())
def pin(p):
    raw=p.read_bytes()
    return dict(bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest())
def same_pin(p,e):
    assert pin(p)=={k:e[k] for k in ('bytes','sha256')},str(p)
def git_bytes(commit,path):
    return subprocess.run(['git','show',f'{commit}:{path}'],cwd=wt,capture_output=True,check=True).stdout

head=subprocess.run(['git','rev-parse','HEAD'],cwd=wt,capture_output=True,text=True,check=True).stdout.strip()
assert head=='2487bfcd73a1463d135a4544a01535ba5c99cab0'
baseline=read(b/'baseline.json')
for name,sha in baseline['sha256'].items():
    assert pin(wt/name)['sha256']==sha
    assert hashlib.sha256(git_bytes(baseline['commit'],name)).hexdigest()==sha
db=sqlite3.connect((wt/'build/hgm_spine_v27_2.db').as_uri()+'?mode=ro',uri=True)
db.row_factory=sqlite3.Row
db.execute('PRAGMA query_only=ON')
sources=[dict(r) for r in db.execute('SELECT * FROM corpus_segments WHERE course=? AND seq BETWEEN ? AND ? ORDER BY seq',('C05',268,270))]
db.close()
assert sources==read(b/'reconciled/268/source-full-rows.json')
assert [{k:r[k] for k in ('id','course','seq','acip','wylie','english')} for r in sources]==read(b/'source.json')
root_originals=read(b/'proposal-verification/root-resolved-spans.json')
root_finals=read(out/'resolved-spans.json')
assert len(root_originals)==227 and len(root_finals)==123
original_by_key={(x['angle'],x['seq'],x['id']):x for x in root_originals}
assert len(original_by_key)==len(root_originals)
frozen=[]; dispositions=Counter(); seen=[]; final_reasons=set(); original_reasons=set()
for seq in (268,269,270):
    d=b/'reconciled'/str(seq); freeze=read(d/'freeze.json')
    expected={e['relative_path'] for e in freeze['files']}
    actual={str(p.relative_to(d)) for p in d.rglob('*') if p.is_file() and p!=d/'freeze.json'}
    assert actual==expected and len(expected)==freeze['file_count']
    assert sum(e['bytes'] for e in freeze['files'])==freeze['total_bytes']
    for e in freeze['files']:
        assert Path(e['path'])==(d/e['relative_path']).resolve()
        same_pin(Path(e['path']),e)
    frozen.append(dict(seq=seq,files=len(expected),freeze=pin(d/'freeze.json')))
    spec=read(d/'spec.json'); seg=spec['segments'][0]
    final=[x for x in root_finals if x['seq']==seq]
    assert read(d/'resolved-spans.json')==final
    assert all(set(x)==set(key7) for x in final)
    decisions=read(d/'final-decisions.json')
    assert [x['final_tuple'] for x in decisions]==final
    assert [x['final_span'] for x in decisions]==seg['spans']
    final_reasons.update(x['rationale'] for x in decisions)
    for x in decisions:
        assert x['rationale'] and x['independent_basis']
        for path in x['evidence']: assert (d/path).is_file(),path
    source=next(x for x in sources if x['seq']==seq)
    gaps=read(d/'unwrapped-extents.json')
    for field,range_key in [('wylie','tib_range'),('english','eng_range')]:
        text=source[field]; covered=[False]*len(text)
        for t in final:
            r=t[range_key]
            if r is None: continue
            a,z=r; assert 0<=a<z<=len(text)
            assert text[a:z]==t['tib' if field=='wylie' else 'eng']
            covered[a:z]=[True]*(z-a)
        actual_gaps=[];i=0
        while i<len(text):
            if covered[i]:i+=1;continue
            j=i+1
            while j<len(text) and not covered[j]:j+=1
            actual_gaps.append(dict(range=[i,j],text=text[i:j]));i=j
        assert actual_gaps==gaps[field]
    for x in read(d/'original-dispositions.json'):
        original=x['original_identity'];assert set(original)==set(key8)
        k=(original['angle'],original['seq'],original['id']);seen.append(k)
        r=original_by_key[k];assert original=={a:r[a] for a in key8}
        p=b/original['angle']/str(seq)/'spec.json'
        span=next(t for t in read(p)['segments'][0]['spans'] if t['id']==original['id'])
        assert x['original_span']==span==r['original_span']
        assert x['original_span_keys']==list(span)
        assert x['original_field_presence']==sorted(span)==r['original_field_presence']
        assert Path(x['original_spec_path'])==p.resolve()
        assert x['reason'];original_reasons.add(x['reason'])
        disposition=x['disposition'];dispositions[disposition]+=1
        assert disposition in ('retained','recast','rejected')
        for f in x['final_identities']:assert f in final
        if disposition=='retained':
            assert len(x['final_identities'])==1
            f=x['final_identities'][0]
            assert {a:original[a] for a in key7 if a!='id'}=={a:f[a] for a in key7 if a!='id'}
        elif disposition=='rejected':assert not x['final_identities']
        else:assert x['final_identities']
assert len(seen)==len(set(seen))==227 and set(seen)==set(original_by_key)
assert dispositions==Counter(retained=200,recast=26,rejected=1)
shared=b/'reconciled/268'
mappings=[]
for name in ('original-to-copy-manifest.json','supplemental-to-copy-manifest.json','helper-and-skill-copy-manifest.json'):
    mappings+=read(shared/name)
mappings+=read(shared/'freeze-packaging-addendum.json')['additional_source_to_copy_mappings']
assert len(mappings)==221
whitelist={'.json','.md','.html','.stderr','.stdout','.log','.txt','.bin','.exit'}
historical=0
for e in mappings:
    copy=shared/e['copy'];same_pin(copy,e);assert copy.suffix in whitelist
    if 'original_git_commit' in e:
        raw=git_bytes(e['original_git_commit'],e['repository_relative_path']);historical+=1
    else:raw=Path(e['original']).read_bytes()
    assert copy.read_bytes()==raw and len(raw)==e['bytes'] and hashlib.sha256(raw).hexdigest()==e['sha256']
assert historical==10
status=subprocess.run(['git','status','--short'],cwd=wt,capture_output=True,text=True,check=True).stdout
assert status=='?? data/teaching/dcc_captions\n'
result=dict(utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),status='PASS mechanical handoff only',head=head,baseline=baseline['commit'],nine_baseline_hashes_equal_live_and_original_Git=True,source_rows_exact=True,reconciliation_freezes=frozen,original_identities=227,dispositions=dict(dispositions),final_tuples=123,nulls=5,nonnull_d5=75,full_original_fields_preserved=True,complete_unwrapped_complements_exact=True,all_final_rationales_and_omissions_read=True,all_original_reason_texts_represented_by_read_final_rationales_or_additional_three=True,additional_original_reason_texts=len(original_reasons-final_reasons),material_mappings=221,historical_Git_pins=historical,all_material_copies_exact=True,all_material_copy_suffixes_supported=True,generator_reports=read(out/'report.json'),scope=status,limitations=['Root authored Tibetan; no independent semantic approval.', 'All three reports,123 final rationales,3 additional original reason texts,omissions and current five-ground rulings read. Earlier full original proposals remain frozen.','Full original physical witnesses and raw query archives are preserved and pinned. Separate reviewer must reopen them independently; root does not claim independent PDF visual review.','Retained probable erratum remains a candidate; no source/master/register/allowance or repository edits.'])
assert result['additional_original_reason_texts']==3
target.write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n')
print(json.dumps({k:v for k,v in result.items() if k not in ['generator_reports','limitations']},ensure_ascii=False))
