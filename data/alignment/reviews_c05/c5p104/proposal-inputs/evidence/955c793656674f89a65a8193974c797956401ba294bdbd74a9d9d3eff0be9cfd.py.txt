for seq,table in tables.items():
    D=A/str(seq); D.mkdir(exist_ok=True)
    spans=[]; intended=[]; parent=None
    for n,(t,e,d,occ,why) in enumerate(table,1):
        sp={'id':'w'+str(n),'d':d,'tib':t,'eng':e}
        if e is None: sp['nul']=why
        spans.append(sp)
        if d!=7: parent=sp['id']
        item={'id':sp['id'],'tib':t,'eng':e,'d':d,'english_occurrence':occ,'rationale':why,'parent':parent if d==7 else None}
        if e is not None:
            text=rows[seq]['english'] if d!=7 else next(v['eng'] for v in spans if v['id']==parent)
            hits=[]; cur=0
            while True:
                at=G.find_word(text,e,cur) if d!=7 else text.find(e,cur)
                if at<0: break
                hits.append(at); cur=at+1
            assert len(hits)>=occ,(seq,t,e,occ,hits)
            lo=hits[occ-1]
            if d==7:
                par=next(v for v in intended if v['id']==parent)
                lo+=par['eng_range'][0]
            item['eng_range']=[lo,lo+len(e)]
            item['all_candidate_starts']=hits
        intended.append(item)
    order=[i['id'] for i in sorted([x for x in intended if x['eng'] is not None and x['d']!=7],key=lambda x:x['eng_range'][0])]
    obj={'course':'C05','segments':[{'seq':seq,'title':{310:'contrasting lives and experiences',311:'the consequence of denial',312:'limits of identical perceptions'}[seq],'spans':spans,'eng_order':order,'note':notes[seq]}]}
    dump(D/'spec.json',obj)
    dump(D/'source-specific-rationale.json',{'status':'authored before first final native generation','seq':seq,'decisions':intended})
    dump(D/'errata.json',[])
    print(seq,len(spans),'authored spans')
