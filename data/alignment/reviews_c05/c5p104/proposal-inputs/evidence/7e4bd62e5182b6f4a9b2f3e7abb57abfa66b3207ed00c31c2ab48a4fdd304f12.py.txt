lex = {'query':'exact wylie or member of wylie_variants in each master list; complete original records, all tiers preserved', 'terms':terms, 'results':{}}
for term in terms:

    for section in ['unified_entries', 'curated_entries', 'glossary_entries']:
        entries = master[section]
        if isinstance(entries, dict): entries = list(entries.values())
        result[section] = [dict(index=i, record=e) for i,e in enumerate(entries) if isinstance(e,dict) and (e.get('wylie') == term or term in (e.get('wylie_variants') or []))]
    lex['results'][term] = result

queries=[]
for row in source:
    for field in ['acip','wylie','english']:
        sql='select * from corpus_segments where '+field+'=? order by course, seq'
        hits=[dict(r) for r in db.execute(sql,(row[field],))]
        queries.append({'key':str(row['seq'])+'-'+field+'-exact','sql':sql,'parameters':[row[field]],'count':len(hits),'rows':hits})

    sql='select * from corpus_segments where '+field+' like ? order by course, seq'
