def make(seq,title,items,note):

   hits=[m.start() for m in re.finditer(re.escape(eng),src['english']) if (m.start()==0 or not src['english'][m.start()-1].isalnum() or not eng[0].isalnum()) and (m.end()==len(src['english']) or not src['english'][m.end()].isalnum() or not eng[-1].isalnum())]
   assert len(hits)>=occ,(seq,eng,occ,hits)
   positions[sid]=hits[occ-1]
  if d!=7:
   t=src['wylie'].find(tib,tc);assert t>=tc,(seq,tib,tc);tc=t+len(tib)
   checks.append({'id':sid,'tib_start':t,'tib_end':tc,'eng_intended_start':positions.get(sid),'eng_occurrence':occ if eng else None})
