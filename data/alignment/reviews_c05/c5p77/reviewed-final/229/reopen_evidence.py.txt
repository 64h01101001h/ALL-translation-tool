def resolve(spec,src):
 s=spec['segments'][0]; spans=s['spans']; byid={x['id']:x for x in spans}
 tr=G.resolve(src['wylie'],spans,'tib',s['seq'])
 es=G.with_members(spans,[byid[i] for i in s['eng_order']],src['english'])
 er=G.resolve(src['english'],es,'eng',s['seq'])
 return [{**x,'tib_range':tr.get(x['id']),'eng_range':er.get(x['id'])} for x in spans]
def walk(x):
 if isinstance(x,dict):
  yield x
  for v in x.values(): yield from walk(v)
 elif isinstance(x,list):
  for v in x: yield from walk(v)
