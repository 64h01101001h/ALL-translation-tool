def build(seq,title,note):
    data[seq]=dict(seq=seq,title=title,spans=[],intended=[],note=note)
def add(seq,tib,eng,d=5,eo=1,to=None,subword=False,why='Direct contextual lexical correspondence.'):
    a=data[seq]; row=sources[seq]; i=len(a['spans'])+1; sid='w'+str(i)
    sp=dict(id=sid,d=d,tib=tib,eng=eng)
    if subword:sp['subword']=True
    def nth(text,piece,n):
        at=-1
        for k in range(n):
            while True:
                at=text.find(piece,at+1)
                if at<0: break
                end=at+len(piece)
                word=lambda c: bool(re.match(r'[A-Za-z0-9\u0F00-\u0FFF]',c))
                if d==7 or subword or ((not word(piece[0]) or at==0 or not word(text[at-1])) and (not word(piece[-1]) or end==len(text) or not word(text[end]))): break
        assert at>=0,(seq,piece,n)
        return [at,at+len(piece)]
    if to is None:
        if d==7:
            parent=next(x for x in reversed(a['intended']) if x['d']==5)
            at=row['wylie'].find(tib,parent['tib_range'][0],parent['tib_range'][1]); tr=[at,at+len(tib)]
        else:
            cur=max([x['tib_range'][1] for x in a['intended'] if x['d']!=7] or [0])
            at=row['wylie'].find(tib,cur);assert at>=0,(seq,tib,cur);tr=[at,at+len(tib)]
    else:tr=nth(row['wylie'],tib,to)
    er=nth(row['english'],eng,eo)
    a['spans'].append(sp);a['intended'].append(dict(id=sid,d=d,tib=tib,eng=eng,tib_range=tr,eng_range=er,reason=why))
    return sid
