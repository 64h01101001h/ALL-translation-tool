import hashlib,json,os,subprocess,sys
from pathlib import Path
H=Path(__file__).resolve().parent
label=sys.argv[1]; cmd=sys.argv[2:]
d=H/'native-commands'/label;d.mkdir(parents=True,exist_ok=True)
idx=1
while (d/str(idx)).exists():idx+=1
a=d/str(idx);a.mkdir()
for i,arg in enumerate(cmd):
    p=Path(arg)
    if p.is_file() and p.suffix=='.py': (a/f'original-arg-{i}-{p.name}').write_bytes(p.read_bytes())
env=dict(os.environ,PYTHONDONTWRITEBYTECODE='1')
r=subprocess.run(cmd,capture_output=True,env=env)
(a/'stdout').write_bytes(r.stdout);(a/'stderr').write_bytes(r.stderr);(a/'exit.txt').write_text(str(r.returncode)+'\n')
(a/'command.json').write_text(json.dumps(dict(argv=cmd,cwd=os.getcwd(),env={'PYTHONDONTWRITEBYTECODE':'1'},exit=r.returncode,stdout_bytes=len(r.stdout),stderr_bytes=len(r.stderr),stdout_sha256=hashlib.sha256(r.stdout).hexdigest(),stderr_sha256=hashlib.sha256(r.stderr).hexdigest()),indent=2)+'\n')
sys.stdout.buffer.write(r.stdout);sys.stderr.buffer.write(r.stderr)
raise SystemExit(r.returncode)
