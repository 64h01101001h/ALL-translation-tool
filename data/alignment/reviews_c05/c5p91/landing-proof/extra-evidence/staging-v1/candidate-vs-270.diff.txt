+                assert int(self.raw(p.parent / (label + '.exit'))) == run['exit']
+                actual_builds.append(dict(record=str(p), label=label, exit=run['exit'], started_at=run['started_at'], finished_at=run['finished_at']))
+        assert len(build['runs']) == 2 and all(r['exit'] == 0 for r in build['runs'])
+        coverage = re.findall(rb'^COVERAGE C05: (\d+)/511 segments scanned, (\d+) open \(([^)]+)\)$', self.raw(self.l / 'build-layer.stdout'), re.M)
+        assert coverage == [(str(SEQS[-1]).encode(), str(511 - SEQS[-1]).encode(), f'{SEQS[-1]+1}-511'.encode())]
+        test_records = []
+        assert {p.parent for p in self.l.rglob('ctest-17.command.json')} == {p.parent for p in self.l.rglob('ctest-17.execution.json')}
+        for execution in sorted(self.l.rglob('ctest-17.execution.json')):
+            record = self.obj(execution); command = self.obj(execution.with_name('ctest-17.command.json'))
+            assert record['argv'] == command['argv'] and record['cwd'] == command['cwd']
+            stdout = self.raw(execution.with_name('ctest-17.stdout')); stderr = self.raw(execution.with_name('ctest-17.stderr'))
+            code = int(self.raw(execution.with_name('ctest-17.exit')))
+            assert record['exit'] == code and record['stdout_bytes'] == len(stdout) and record['stderr_bytes'] == len(stderr)
+            passed = re.findall(r'^\s*(\d+)/(\d+) Test\s+#(\d+): ([a-z_]+) .* Passed\s+([0-9.]+) sec$', stdout.decode(), re.M)
+            duration = re.findall(r'^Total Test time \(real\) =\s*([0-9.]+) sec$', stdout.decode(), re.M)
+            if execution.parent == self.l:
+                assert code == 0 and not stderr and len(passed) == len(TESTS)
+                assert {r[3] for r in passed} == TESTS
+                assert [int(r[0]) for r in passed] == list(range(1, len(TESTS) + 1))
+                assert all(int(r[1]) == len(TESTS) for r in passed) and len(duration) == 1
+                assert f'100% tests passed out of {len(TESTS)}' in stdout.decode()
+            test_records.append(dict(execution=str(execution), exit=code, actual_passed_records=passed, actual_total_seconds=duration, recorded_wall_seconds=record['wall_seconds']))
+        assert any(r['execution'] == str(self.l / 'ctest-17.execution.json') for r in test_records)
+
+        # Root proof bank is a separate trust boundary, never semantic review text.
+        root_manifest_raw = self.raw(self.l / 'root-proof-copy-manifest.json')
+        root_manifest = json.loads(root_manifest_raw)
+        assert root_manifest['semantic_review_unchanged'] is True
+        for row in root_manifest['files']:
+            raw = self.raw(Path(row['original'])); self.pincheck(raw, row)
+            self.retained_originals.add(str(Path(row['original']).resolve()))
+            self.require(self.bank_rel / 'landing-proof' / relative(row['copy']), raw)
+        for row in root_manifest['external_actual_bytes_retained']:
+            raw = self.raw(Path(row['path'])); self.pincheck(raw, row)
+            self.retained_originals.add(str(Path(row['path']).resolve()))
+        self.require(self.bank_rel / 'landing-proof/root-proof-copy-manifest.json', root_manifest_raw)
+        previous_manifest = self.l / 'root-proof-copy-manifest-before-whitespace.json'
+        self.require(self.bank_rel / 'landing-proof' / previous_manifest.name, self.raw(previous_manifest))
+        for p in self.l.rglob('*'):
+            if p.is_file() and p not in (self.l / 'root-proof-copy-manifest.json', previous_manifest):
+                assert str(p.resolve()) in self.retained_originals, ('unretained landing evidence', p)
+        assert str(Path(self.args.untracked_before).resolve()) in self.retained_originals
+
+        # Only consume later actual whitespace proof; no exception classes here.
+        ws_path = Path(self.args.whitespace_proof)
+        whitespace = self.obj(ws_path)
+        assert str(ws_path.resolve()) in self.retained_originals
+        assert whitespace['baseline_head'] == BASE and whitespace['review_freeze_sha256'] == sha(freeze_raw)
+        assert whitespace['all_warnings_accounted_for'] is True
+        raw_log = self.raw(Path(self.args.whitespace_log))
+        assert str(Path(self.args.whitespace_log).resolve()) in self.retained_originals
+        assert len(raw_log) == whitespace['raw_log_bytes'] and sha(raw_log) == whitespace['raw_log_sha256']
+        diagnostic, code = self.git('diff', '--cached', '--check', allowed=(0, 2))
+        assert diagnostic == raw_log and code == whitespace['git_diff_check_exit']
+        if code == 0:
+            assert not diagnostic and whitespace['warning_count'] == 0
+        else:
+            assert diagnostic and whitespace['warning_count'] > 0
+        for rel, row in whitespace['files'].items():
+            self.pincheck(self.raw(self.w / relative(rel)), row)
+
+        # Full page shell and optional generated docs also require earlier frozen bytes.
+        payload_path = Path(self.args.payload_manifest)
+        payload = self.obj(payload_path)
+        assert str(payload_path.resolve()) in self.retained_originals
+        assert payload['head'] == BASE
+        payload_records = {}
+        for row in payload['files']:
+            rel = str(relative(row['path']))
+            assert rel not in payload_records
+            payload_records[rel] = row
+        payload_required = {rel for rel in self.required if not Path(rel).is_relative_to(self.bank_rel)}
+        assert set(payload_records) == payload_required
+        for rel, row in payload_records.items():
+            self.pincheck(self.required[rel], row)
+
+        # No broad path allowance: derive exact staged set from actual pinned files.
+        assert set(staged) == set(self.required), (set(staged) - set(self.required), set(self.required) - set(staged))
+        staged_files = []
+        for rel in staged:
+            raw = self.required[rel]
+            assert self.git('show', ':' + rel)[0] == raw
+            assert self.raw(self.w / rel) == raw
+            staged_files.append(dict(path=rel, bytes=len(raw), sha256=sha(raw)))
+        assert self.git('ls-files', '--stage', '-z')[0] == index_before
+        assert not self.git('diff', '--name-only', '-z')[0]
+        assert self.git('ls-files', '--others', '--exclude-standard', '-z')[0] == initial_untracked
+        for p, pin in self.inputs.items():
+            self.pincheck(Path(p).read_bytes(), pin)
+        return dict(status='PASS', checked_at=datetime.datetime.now(datetime.timezone.utc).isoformat(),
+                    head=BASE, page=PAGE, segments=SEQS, actual_review_totals=totals,
+                    staged_count=len(staged_files), files=staged_files, source_hashes_unchanged=True,
+                    actual_build_invocations=actual_builds, actual_test_executions=test_records,
+                    actual_coverage=int(coverage[0][0]), actual_added_links=len(added),
+                    whitespace_exit=code, whitespace_bytes=len(raw_log), whitespace_sha256=sha(raw_log),
+                    semantic_frozen_files=len(frozen), no_semantic_approval_supplied=True,
+                    proof_boundary='Exact current stage before this new external output; any later banking/stage change needs another fresh boundary.')
+

+def main():
+    parser = argparse.ArgumentParser(description=__doc__)
+    for name in ('worktree', 'batch', 'output', 'untracked-before', 'whitespace-proof', 'whitespace-log', 'payload-manifest'):
+        parser.add_argument('--' + name, required=True)
+    args = parser.parse_args()
+    if not __debug__:
+        raise RuntimeError('Assertions must remain enabled; do not use optimized Python')
+    check = Check(args)
+    (check.out / 'invocation.json').write_text(json.dumps(dict(argv=sys.argv, cwd=os.getcwd(), environment=dict(PYTHONDONTWRITEBYTECODE=os.environ.get('PYTHONDONTWRITEBYTECODE'))), indent=2) + '\n')
+    try:
+        proof = check.run()
+    except Exception:
+        (check.out / 'failure.txt').write_text(traceback.format_exc())
+        raise
+    finally:
+        (check.out / 'consumed-input-pins.json').write_text(json.dumps(check.inputs, indent=2) + '\n')
+    (check.out / 'root-final-stage-verification.json').write_text(json.dumps(proof, indent=2) + '\n')
+    print(json.dumps({k: v for k, v in proof.items() if k != 'files'}, indent=2))
+
+
+if __name__ == '__main__':
+    main()
