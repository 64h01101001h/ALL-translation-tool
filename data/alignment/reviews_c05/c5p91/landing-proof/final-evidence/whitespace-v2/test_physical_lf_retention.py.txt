import unittest
from whitespace273 import classify_physical
class Retention(unittest.TestCase):
 def test_preexisting_source_owned_lf_mixed_warnings_remain_valid(self):
  raw=b'A \n\n';r=classify_physical(raw,{'trailing whitespace.':[1],'new blank line at EOF.':[2]},[({'path':'TOOLING_ONLY'},b'prefix\n'+raw)])
  self.assertEqual(r['kind'],'physical_bytes');self.assertEqual(r['full_range_witnesses'][0]['byte_range'],[7,11])
if __name__=='__main__':unittest.main(verbosity=2)
