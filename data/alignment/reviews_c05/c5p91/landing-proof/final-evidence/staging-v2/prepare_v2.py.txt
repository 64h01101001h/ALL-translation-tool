def allowance_result(old_raw, new_raw, wanted, registered):
    allowed = json.loads(old_raw)
    if isinstance(wanted, list):
        normalized = {}
        for row in wanted:
            category, key = row['key'].split('/', 1) if 'key' in row else (row['pages_directory'], row['allowlist_key'])
            assert key not in normalized.setdefault(category, {})
            normalized[category][key] = row['licensor']
        wanted = normalized
    assert isinstance(wanted, dict) and wanted == registered
    additions = 0
    for category, rows in wanted.items():
        assert isinstance(rows, dict) and category in allowed
        for key, value in rows.items():
            assert key not in allowed[category]
            allowed[category][key] = value
            additions += 1
    assert json.loads(new_raw) == allowed
    if not additions:
        assert new_raw == old_raw, 'Zero additions require complete baseline byte identity'
    return wanted, bool(additions)


def nul_paths(raw):
    assert not raw or raw.endswith(b'\\0'), 'Incomplete raw Git path stream'
    values = [os.fsdecode(value) for value in raw.split(b'\\0')[:-1]] if raw else []
    assert len(values) == len(set(values))
    for value in values:
        relative(value)
    return values


def untracked_transition(before, after, newly_staged, root_bank_paths, later_written):
    initial = nul_paths(before)
    current = nul_paths(after)
    newly_staged, root_bank_paths, later_written = map(set, (newly_staged, root_bank_paths, later_written))
    assert root_bank_paths <= newly_staged
    assert not set(initial) & root_bank_paths, 'Capture must precede ordinary bank creation'
    assert newly_staged - root_bank_paths <= set(initial), 'New non-bank payload absent from true pre-stage capture'
    assert later_written == root_bank_paths, 'Every later root-bank path needs exact actual write evidence'
    removed = set(initial) & newly_staged
    residual = [value for value in initial if value not in removed]
    expected_raw = b''.join(os.fsencode(value) + b'\\0' for value in residual)
    assert after == expected_raw, ('Unrelated untracked drift', set(current) - set(residual), set(residual) - set(current))
    return dict(initial_paths_newly_staged=sorted(removed),
                later_created_root_bank_paths=sorted(root_bank_paths), residual_paths=residual,
                raw_before_bytes=len(before), raw_before_sha256=sha(before),
                raw_after_bytes=len(after), raw_after_sha256=sha(after),
                boundary='Raw capture before ordinary banking and before any staging; bank paths created later are not claimed in that stream')
