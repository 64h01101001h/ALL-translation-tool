from pathlib import Path
import ast,json,unittest
D=Path(__file__).resolve().parent
p=D/'inputs/staging-helper/verify_staged_through273.py'
tree=ast.parse(p.read_text()); run=next(n for c in tree.body if isinstance(c,ast.ClassDef) and c.name=='Check' for n in c.body if isinstance(n,ast.FunctionDef) and n.name=='run')
def block(first,last,g):
 exec(compile(ast.Module(body=[n for n in run.body if first<=n.lineno<=last],type_ignores=[]),str(p),'exec'),g)
class Tests(unittest.TestCase):
 def test_actual_approved_freeze_should_be_accepted(self):
  f=json.loads((D/'inputs/schema/freeze.json').read_bytes())
  try:block(123,123,{'freeze':f})
  except Exception as e:self.fail('Actual SPEC/QUALITY APPROVE rejected: '+repr(e))
 def test_empty_category_should_not_require_staged_delta(self):
  calls=[]
  class Capture:
   def require(self,*a):calls.append(a)
  block(268,268,dict(wanted={'pages_c05':{}},self=Capture(),rel='allow',old_raw=b'{}\n',new_raw=b'{}\n'))
  self.assertEqual(calls,[])
 def test_authorized_staging_should_preserve_only_unrelated_untracked(self):
  class GitRead:
   def git(self,*a):return b'unrelated.txt\0',0
  try:block(431,431,dict(self=GitRead(),initial_untracked=b'new-page.txt\0unrelated.txt\0'))
  except AssertionError:self.fail('Authorized new-page staging incorrectly rejected')
if __name__=='__main__':unittest.main(verbosity=2)
