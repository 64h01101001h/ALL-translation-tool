def read(p): return json.loads(p.read_bytes())
def pin(p):
    raw = p.read_bytes()
    return dict(bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest())
def checkpin(p, expected):
    assert pin(p) == {k: expected[k] for k in ('bytes', 'sha256')}, str(p)
def save(p, obj):
    with p.open('x') as f: json.dump(obj, f, ensure_ascii=False, indent=2); f.write('\n')
def git(*args):

def freeze(directory):
    fp = directory / 'freeze.json'; f = read(fp)
    items = f['files'].items() if isinstance(f['files'], dict) else [(r.get('relative_path', r.get('path')), r) for r in f['files']]
    entries = {}
    for name, record in items:
        path = Path(name)
        name = str(path.relative_to(directory)) if path.is_absolute() else name
        assert not Path(name).is_absolute() and '..' not in Path(name).parts and name not in entries
        if Path(record.get('path', '')).is_absolute(): assert Path(record['path']) == directory/name
        entries[name] = record
    actual = {str(p.relative_to(directory)) for p in directory.rglob('*') if p.is_file() and p != fp}
    assert actual == set(entries), (directory, actual ^ set(entries))
    if 'file_count' in f: assert len(entries) == f['file_count']
    for name, record in entries.items(): checkpin(directory/name, record)
    if 'total_bytes' in f: assert sum(r['bytes'] for r in entries.values()) == f['total_bytes']
    return dict(directory=str(directory), files=len(entries), freeze=pin(fp))
