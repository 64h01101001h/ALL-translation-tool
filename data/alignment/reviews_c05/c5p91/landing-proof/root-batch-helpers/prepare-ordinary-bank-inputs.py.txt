"""Inventory complete finished evidence trees for the approved ordinary bank.

Every frozen member is retained; selected small readable evidence is also copied.
No source, index, or repository writes occur here.
"""
from pathlib import Path
import datetime, hashlib, json

B=Path(__file__).resolve().parent;A=B.parent;I=A/'integration';L=B/'landing'
TARGET=L/'banking-ordinary-extra-inputs.json';PROOF=L/'banking-ordinary-extra-inputs-proof.json'
assert not TARGET.exists() and not PROOF.exists()
records=[];trees=[]
def pin(p):
    x=p.read_bytes();return dict(bytes=len(x),sha256=hashlib.sha256(x).hexdigest())
def add(p,copy=None):
    rec=dict(original=str(p.resolve()),**pin(p))
    if copy is not None:rec.update(retention='bank',copy=copy+('.txt' if p.suffix=='.py' else ''))
    else:rec.update(retention='external',disposition='Complete frozen original evidence retained at this exact path; full tree inventory and every nested failed version preserved. No shortening or normalization.')
    records.append(rec)
def frozen_tree(folder,label):
    f=json.loads((folder/'freeze.json').read_bytes());items=f['files']
    expected=items if isinstance(items,dict) else {r['path']:{k:r[k] for k in ('bytes','sha256')} for r in items}
    actual={str(p.relative_to(folder)):pin(p) for p in folder.rglob('*') if p.is_file() and p!=folder/'freeze.json'}
    assert actual==expected
    trees.append(dict(root=str(folder),label=label,freeze=pin(folder/'freeze.json'),files=len(actual),bytes=sum(v['bytes'] for v in actual.values())))
    for p in sorted(folder.rglob('*')):
        if not p.is_file():continue
        rel=p.relative_to(folder)
        readable=(len(rel.parts)==1 and p.stat().st_size<2_000_000) or (rel.parts[0] in ('history','executions') and p.stat().st_size<200_000)
        add(p,'extra-evidence/'+label+'/'+str(rel) if readable else None)
for folder,label in [
    (I/'through273-generated-reproduction','generated-reproduction'),
    (I/'through273-preparation/banking-helper','banking-tool'),
    (I/'through273-preparation/staging-helper','staging-v1'),
    (I/'through273-preparation/staging-review','staging-v1-review')]:
    frozen_tree(folder,label)
for folder,label in [
    (I/'through273-generated-reproduction-root','generated-reproduction-root'),
    (I/'through273-preparation/banking-root-review','banking-root-review'),
    (I/'through273-preparation/staging-review-root','staging-review-root'),
    (B/'whitespace-preparation','whitespace-static-preparation')]:
    for p in sorted(folder.rglob('*')):
        if p.is_file():add(p,'extra-evidence/'+label+'/'+str(p.relative_to(folder)))
for name in ('verify_semantic_acceptance.py','land_reviewed_page.py','verify_review_copy_and_register.py',
             'run_reviewed_builders_once.py','verify_landed_batch.py','verify_course_csv.py',
             'run_tool_readonly_sqlite.py','run_generator_readonly.py','copy_evidence_once_v2.py','record_external_step.py'):
    add(A/name,'root-shared-helpers/'+name)
add(I/'verify-through273-reproduction-root.py','root-shared-helpers/verify-through273-reproduction-root.py')
for p in sorted(B.glob('*.py')):add(p,'root-batch-helpers/'+p.name)
for name in ('baseline.json','source.json'):add(B/name,'root-batch-inputs/'+name)
assert len({(r['original'],r.get('copy')) for r in records})==len(records)
TARGET.write_text(json.dumps(records,indent=2)+'\n')
result=dict(status='PASS complete explicit ordinary extra inventory',utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),
    trees=trees,records=len(records),banked=sum(r['retention']=='bank' for r in records),
    external=sum(r['retention']=='external' for r in records),inventory=dict(path=str(TARGET),**pin(TARGET)),
    scope='All actual completed frozen members including top freezes plus explicit root helper/proof inputs; no pending v2 staging/whitespace tree included. Later finished evidence requires final phase inventory.')
PROOF.write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result,indent=2))
