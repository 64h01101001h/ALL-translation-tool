from pathlib import Path
from datetime import datetime, timezone
import ast, difflib, hashlib, json
C=Path(__file__).resolve().parent
B=C.parent
R=C/'reconciler'
def j(p): return json.loads(p.read_text())
def pin(p):
    raw=p.read_bytes();return {'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}
def check(p,x): assert pin(p)=={k:x[k] for k in ('bytes','sha256')}, str(p)
def inventory(p): return {str(x.relative_to(p)):pin(x) for x in sorted(p.rglob('*')) if x.is_file()}

freezes=[]
for p in (R,B/'reconciled/273'):
    actual=inventory(p);f=j(p/'freeze.json')
    assert {k:v for k,v in actual.items() if k!='freeze.json'}==f['files']
    freezes.append({'path':str(p/'freeze.json'),**pin(p/'freeze.json'),'files':len(f['files']),'total_bytes':sum(x['bytes'] for x in f['files'].values())})

    assert {k:v for k,v in before['trees']['reconciled'].items() if k.startswith(seq)}=={k:v for k,v in after['trees']['reconciled'].items() if k.startswith(seq)}
rows=j(R/'all-source-copy-byte-checks.json');assert len(rows)==412

    check(Path(x['original']),x);check(Path(x['copy']),x)
    assert Path(x['original']).read_bytes()==Path(x['copy']).read_bytes() and x['full_bytes_equal']

    current=Path(x['original_active_path']);assert current.is_relative_to(B/'reconciled/273')
    check(current,x['after'])
    if x['kind']=='changed':
        assert x['before'] is not None;check(Path(x['initial_retained_copy']),x['before'])
    else: assert x['kind']=='added' and x['before'] is None and x['initial_retained_copy'] is None
diff=[]
for k in changed:
    a=C/'initial/reconciled'/k;z=B/'reconciled'/k
    diff.extend(difflib.unified_diff(a.read_text().splitlines(True) if a.exists() else [],z.read_text().splitlines(True),fromfile=str(a) if a.exists() else '/dev/null',tofile=str(z)))

assert allowed in sets
assert all(Path(x['copy']).suffix in allowed for x in rows)
