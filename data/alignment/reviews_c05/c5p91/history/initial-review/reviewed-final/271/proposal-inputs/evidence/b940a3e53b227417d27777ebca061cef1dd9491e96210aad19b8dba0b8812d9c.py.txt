def hits(text,piece):
    return [[m.start(),m.end()] for m in re.finditer(r'(?<![A-Za-z0-9])'+re.escape(piece)+r'(?![A-Za-z0-9])',text)]

class Draft:
    def __init__(self,seq,title):
        self.seq=seq;self.row=ROWS[seq];self.title=title;self.spans=[];self.decisions=[];self.cursor=0;self.parent=None
    def add(self,tib,eng,d=5,n=1,context=None,reason='',nul=None):
        identity={3:'r',5:'w',6:'p',7:'m'}[d]+str(len(self.spans)+1)
        if d==7:
            parent,pr=self.parent
            assert parent['tib'].count(tib)==1 and parent['eng'].count(eng)==1
            tr=[pr['tib_range'][0]+parent['tib'].index(tib),pr['tib_range'][0]+parent['tib'].index(tib)+len(tib)]
            er=[pr['eng_range'][0]+parent['eng'].index(eng),pr['eng_range'][0]+parent['eng'].index(eng)+len(eng)]
            assert tib!=parent['tib'] and eng!=parent['eng']
        else:
            possible=[r for r in hits(self.row['wylie'],tib) if r[0]>=self.cursor]
            assert possible,(self.seq,tib,self.cursor)
            tr=possible[0];self.cursor=tr[1]
            er=None
            if eng is not None:
                choices=hits(self.row['english'],eng)
                if context:
                    start=self.row['english'].index(context);end=start+len(context)
                    choices=[r for r in choices if start<=r[0] and r[1]<=end]
                assert len(choices)>=n,(self.seq,eng,n,choices)
                er=choices[n-1]
        sp=dict(id=identity,d=d,tib=tib,eng=eng)
        if eng is None:
            assert nul
            sp.update(cls='grammar',nul=nul)
        decision=dict(id=identity,tib_range=tr,eng_range=er,tib_occurrences=hits(self.row['wylie'],tib),
            eng_occurrences=hits(self.row['english'],eng) if eng is not None else [],reason=reason,
            source_role=reason.split(';')[0],null_rationale=nul,parent=self.parent[0]['id'] if d==7 else None)
        self.spans.append(sp);self.decisions.append(decision)
        if d==5:self.parent=(sp,decision)
        elif d!=7:self.parent=None
        return identity
    def save(self,note,omissions):
        flat=[(d['eng_range'][0],s['id']) for s,d in zip(self.spans,self.decisions) if s['eng'] is not None and s['d']!=7]
        seg=dict(seq=self.seq,title=self.title,spans=self.spans,eng_order=[identity for _,identity in sorted(flat)],note=note)
        p=ANGLE/str(self.seq)
        (p/'spec.json').write_text(json.dumps(dict(course='C05',segments=[seg]),ensure_ascii=False,indent=2)+'\n')
        (p/'decisions.json').write_text(json.dumps(dict(producer='Codex; reused English-first original author; no other-angle reads',seq=self.seq,
            occurrence_coordinate_system='zero-based Python Unicode character ranges in the complete original source fields',
            spans=self.decisions,omission_decisions=omissions,not_a_semantic_approval=True),ensure_ascii=False,indent=2)+'\n')
        (p/'errata.json').write_text('[]\n')
