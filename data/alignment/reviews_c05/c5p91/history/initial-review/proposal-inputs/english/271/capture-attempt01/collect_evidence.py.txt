def meta(path):
    raw=path.read_bytes();return dict(path=str(path),bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest())
def put(name,obj):
    p=OUT/name;p.parent.mkdir(parents=True,exist_ok=True)
    with p.open('x') as f:json.dump(obj,f,ensure_ascii=False,indent=2);f.write('\n')
def run(label,argv,stdin=b'',cwd=WT):
    dest=OUT/'commands'/label;dest.mkdir(parents=True)
    (dest/'stdin.bin').write_bytes(stdin)
    put('commands/'+label+'/command.json',dict(argv=argv,cwd=str(cwd),stdin_bytes=len(stdin),environment_overrides={'PYTHONDONTWRITEBYTECODE':'1'}))
    r=subprocess.run(argv,cwd=cwd,env=dict(os.environ,PYTHONDONTWRITEBYTECODE='1'),input=stdin,capture_output=True)
    (dest/'stdout.bin').write_bytes(r.stdout);(dest/'stderr.bin').write_bytes(r.stderr);(dest/'exit.txt').write_text(str(r.returncode)+'\n')
    assert r.returncode==0 and not r.stderr,(label,r.returncode)
    return r.stdout
