"""Collect frozen inputs and complete diffs; does not approve or mutate inputs."""
from pathlib import Path
import difflib
import hashlib
import json
import sys

first, last = map(int, sys.argv[1:])
assert last == first + 2
batch = Path(__file__).resolve().parent / f"C05-{first}-{last}"
target = batch / "semantic-input-review-package.txt"
assert not target.exists(), "Preserve the prior package before rebuilding"
chunks = []
manifest = []

def include(path):
    if path.is_file():
        raw = path.read_bytes()
        chunks.extend([f"\n===== {path.relative_to(batch)} =====\n", raw.decode()])
        manifest.append({"path": str(path.relative_to(batch)), "bytes": len(raw),
                         "sha256": hashlib.sha256(raw).hexdigest()})

for name in ("source.json", "context.json", "root-review-targets.md",
             "root-proposal-version-note.md", "root-reconciliation-verification/report.json"):
    include(batch / name)
for seq in range(first, last + 1):
    for angle in ("tibetan", "english", "reconciled"):
        for name in ("spec.json", "report.md", "decisions.md", "detailed-decisions.md", "proposal-decisions.json",
                     "final-span-decisions.json", "errata-five-ground-verdicts.json",
                     "decision-occurrence-correction.md", "errata.json"):
            include(batch / angle / str(seq) / name)
    for angle in ("tibetan", "english"):
        old = batch / angle / str(seq) / "spec.json"
        new = batch / "reconciled" / str(seq) / "spec.json"
        chunks.extend([f"\n===== COMPLETE {angle} TO RECONCILED DIFF {seq} =====\n",
                       "".join(difflib.unified_diff(
                           old.read_text().splitlines(True), new.read_text().splitlines(True),
                           fromfile=str(old.relative_to(batch)),
                           tofile=str(new.relative_to(batch)), n=10))])
chunks.extend(["\n===== PACKAGE INPUT HASHES =====\n", json.dumps(manifest, indent=2), "\n"])
target.write_text("".join(chunks))
print(json.dumps({"path": str(target), "bytes": target.stat().st_size,
                  "sha256": hashlib.sha256(target.read_bytes()).hexdigest()}))
