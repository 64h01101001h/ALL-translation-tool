def check(test, message):
    if not test: raise ValueError(message)

def read(p): return json.loads(p.read_bytes())
def pin(p):
    h = hashlib.sha256(); n = 0
    with p.open('rb') as stream:
        for data in iter(lambda: stream.read(1048576), b''): n += len(data); h.update(data)
    return dict(bytes=n, sha256=h.hexdigest())
def exp(x): return {k:x[k] for k in ('bytes','sha256')}
def save(p, x):
    check(not p.exists(), 'refuse overwrite: ' + str(p))
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(x, ensure_ascii=False, indent=2)+'\n')
def write_new(p, raw):
    check(not p.exists(), 'refuse overwrite: ' + str(p))
    p.parent.mkdir(parents=True, exist_ok=True); p.write_bytes(raw)

commands = []
def git(*args):
    argv = ['git', '-C', str(W), *args]
    p = subprocess.run(argv, input=b'', stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                       env=dict(os.environ, GIT_OPTIONAL_LOCKS='0'))
    d = F/'commands'/f'{len(commands):03d}'; d.mkdir(parents=True)
    for name, raw in [('stdin',b''),('stdout',p.stdout),('stderr',p.stderr)]: (d/(name+'.bin')).write_bytes(raw)
    (d/'exit.txt').write_text(str(p.returncode)+'\n')
    record = dict(argv=argv,cwd=str(S),exit=p.returncode,
                  streams={k:pin(d/(k+'.bin')) for k in ('stdin','stdout','stderr')})
    save(d/'execution.json',record); commands.append(str(d.relative_to(S)))
    check(p.returncode == 0, 'Git command failed; full streams preserved')
    return p.stdout
