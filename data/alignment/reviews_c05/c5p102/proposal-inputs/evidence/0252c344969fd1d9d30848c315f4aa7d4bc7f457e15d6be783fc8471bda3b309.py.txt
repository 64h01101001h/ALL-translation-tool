"""Verify frozen input identities and actual existing executions; no generation."""
from pathlib import Path
import collections, datetime, gzip, hashlib, json, sqlite3, subprocess

B = Path(__file__).resolve().parent
D = B / 'reconciled'
W = B.parent.parent / 'campaign-worktree'
pin = lambda b: {'bytes': len(b), 'sha256': hashlib.sha256(b).hexdigest()}
read = lambda p: json.loads(p.read_bytes())
out = B / 'root-reconciliation-intake.json'
assert not out.exists()
freezes = {}
for angle, expected in [('tibetan', 'd83b6b839f46e6c7be04106e7f1e065e16282ec45f0fb13233daefcbde9561ca'), ('english', 'f6dcf4aba79de3718a2077f5639b99b46ebf0ee41113e9606235895cd6a43a9c'), ('reconciled', '3212c15bf512c4cd5bc3b12467de2eb85c845f5a768190b7bb9c8f42e1704e94')]:
    p = B / angle / ('freeze.json' if angle == 'reconciled' else 'freeze-manifest.json')
    assert pin(p.read_bytes())['sha256'] == expected
    rows = read(p)['files']; seen = set()
    for r in rows:
        q = B / angle / r['path']
        assert q.resolve().is_relative_to((B / angle).resolve()) and r['path'] not in seen
        seen.add(r['path'])
        assert pin(q.read_bytes()) == {k:r[k] for k in ('bytes','sha256')}
    freezes[angle] = dict(files=len(rows), freeze=pin(p.read_bytes()))
copies = read(D/'originals/copy-manifest.json')
for r in copies:
    raw = Path(r['original']).read_bytes()
    assert raw == (D/r['copy']).read_bytes() and pin(raw) == {k:r[k] for k in ('bytes','sha256')}
for r in read(D/'evidence/independent-source-pins.json'):
    assert pin(Path(r['path']).read_bytes()) == {k:r[k] for k in ('bytes','sha256')}
base = read(B/'proposal-baseline.json')
for rel, r in base['governing_git_pins'].items():
    assert pin(subprocess.check_output(['git','show',r['original_git_commit']+':'+rel],cwd=W)) == {k:r[k] for k in ('bytes','sha256')}
master = json.loads(gzip.decompress((W/'data/hgm_dictionary_v27_2.json.gz').read_bytes()))
material = read(B/'root-material-record-reading-list.json')
for r in material:
    section,index = r['original_location'].split(':')
    assert read(D/'evidence'/r['full_record_ref']) == master[section][int(index)]
db=sqlite3.connect((W/'build/hgm_spine_v27_2.db').as_uri()+'?mode=ro',uri=True)
db.row_factory=sqlite3.Row; db.execute('PRAGMA query_only=ON')
for r in read(B/'source.json'):
    row=dict(db.execute('select * from corpus_segments where course=? and seq=?',(r['course'],r['seq'])).fetchone())
    assert all(row[k]==v for k,v in r.items())
for r in read(D/'evidence/attribution-quoted-witnesses.json').values():
    if isinstance(r,dict):
        assert r==dict(db.execute('select * from corpus_segments where course=? and seq=?',(r['course'],r['seq'])).fetchone())
db.close()
counts=[]; dispositions=collections.Counter(); annotation_changes=[]; checked=0
resolved=read(B/'root-reconciliation-verification/resolved-spans.json')
for seq in (304,305,306):
    spans=read(D/str(seq)/'spec.json')['segments'][0]['spans']; byid={s['id']:s for s in spans}
    rr=[r for r in resolved if r['seq']==seq]
    intended=read(D/str(seq)/'final-ordered-tuples.json')['tibetan_order']
    assert len(rr)==len(intended)==len(spans)
    for r,x in zip(rr,intended):
        assert all(r[k]==x[k] for k in ('id','d','tib','eng','tib_range','eng_range'))
    for angle in ('tibetan','english'):
        originals=read(B/angle/str(seq)/'spec.json')['segments'][0]['spans']
        ds=[r for r in read(D/str(seq)/'original-span-dispositions.json') if r['angle']==angle]
        assert [r['original_span'] for r in ds]==originals
        for r in ds:
            dispositions[r['disposition']]+=1; checked+=1
            if r['disposition']=='retained':
                assert len(r['final_ids'])==1
                s=byid[r['final_ids'][0]]; original=r['original_span']
                assert all(s[k]==original[k] for k in ('d','tib','eng'))
                final=next(x for x in rr if x['id']==s['id'])
                assert final['tib_range']==r['original_actual_tib_range'] and final['eng_range']==r['original_actual_eng_range']
                old={k:v for k,v in original.items() if k!='id'}; new={k:v for k,v in s.items() if k!='id'}
                if old!=new:
                    delta={k:{'original':old.get(k),'final':new.get(k)} for k in set(old)|set(new) if old.get(k)!=new.get(k)}
                    assert set(delta)<= {'cls','nul'}
                    annotation_changes.append(dict(seq=seq,angle=angle,original_id=r['original_id'],final_id=s['id'],changed_fields=delta))
        native=D/f'native/original-{angle}-{seq}'; e=read(native/'execution.json')
        assert e['exit']==0 and (native/'exit.txt').read_bytes()==b'0\n' and not (native/'stderr.bin').read_bytes()
        for stream,r in e['streams'].items(): assert pin((native/(stream+'.bin')).read_bytes())==r
        assert (native/'stdin.bin').read_bytes()==(B/angle/str(seq)/'spec.json').read_bytes()
        assert (native/'stdout.bin').read_bytes()==(B/angle/str(seq)/'body.html').read_bytes()
    counts.append(dict(seq=seq,spans=len(spans),nulls=sum(s['eng'] is None for s in spans),d5_total=sum(s['d']==5 for s in spans),nonnull_d5=sum(s['d']==5 and s['eng'] is not None for s in spans)))
assert checked==141 and len(annotation_changes)==3
result=dict(status='PASS_ROOT_INTAKE_NOT_FRESH_SEMANTIC_ACCEPTANCE',utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),freezes=freezes,original_copy_records_exact=len(copies),source_pins_and_historical_governance_exact=True,full_material_master_records_read_and_matched=len(material),counts=counts,original_dispositions=dict(dispositions),retained_tuple_annotation_changes=annotation_changes,
    original_native_runs='Reconciler executed six once; root verified actual complete native streams/command records. No redundant root original generation.',
    root_final_native_runs='Three actual current canonical generators passed; all74 source/English ranges equal saved intents.',
    root_read='Complete six original specs/reports/errata and grouped decisions; all141 dispositions/material reasons; all74 final tuples/reasons/notes; omissions/nulls/five-ground screens;25 complete material master records; three complete selected source rows and two full attribution witnesses. One truncated final-reason display recovered in a bounded read.',
    count_clarification='44 total d5 spans includes one null yin in304. Consumer word_pair_count is43, not44. Counts by segment7/15/21 nonnull d5.',
    setup_failure='Initial root inline intake wrongly treated retained tuple as byte-identical entire annotation object; assertion refused before proof write on306 English p1. Exact d/tib/eng/source ranges remain retained. Three changed null annotations, including zhes cls particle to case, are now explicitly inventoried for independent review. No source/generator or tests repeated.',
    limitations='Root read25 selected material master objects, not every broad query or surrounding ILL context. Fresh skeptic must read all material originals and source argument. Historical governance ce646 through294 remains separate from actual future acceptance after303.')
out.write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n')
print(json.dumps({k:result[k] for k in ('status','counts','original_dispositions','count_clarification')},indent=2))
