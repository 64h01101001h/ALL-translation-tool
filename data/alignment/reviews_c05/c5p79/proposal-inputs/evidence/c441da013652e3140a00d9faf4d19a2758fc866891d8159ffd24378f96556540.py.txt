def write_json(path: Path, value: object) -> None:
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def rows_as_dicts(cursor: sqlite3.Cursor, rows: list[tuple]) -> list[dict]:
    columns = [item[0] for item in cursor.description]
    return [dict(zip(columns, row)) for row in rows]


def main() -> None:
    snapshot_rows = {row["seq"]: row for row in json.loads(SNAPSHOT.read_text(encoding="utf-8"))}
    connection = sqlite3.connect(DB_URI, uri=True)
    connection.execute("PRAGMA query_only=ON")

    cursor = connection.execute(
        "SELECT id,course,seq,acip,wylie,english FROM corpus_segments "
        "WHERE course=? AND seq BETWEEN ? AND ? ORDER BY seq",
        ("C05", 233, 239),
    )
    context_rows = rows_as_dicts(cursor, cursor.fetchall())
    by_seq = {row["seq"]: row for row in context_rows}

    for seq in SEQS:
        out = BASE / str(seq)
        source_row = by_seq[seq]
        write_json(out / "source-original.json", source_row)
        write_json(out / "context-original.json", context_rows)

        comparisons = {}
        for field in ("id", "course", "seq", "acip", "wylie", "english"):
            comparisons[field] = {
                "snapshot": snapshot_rows[seq][field],
                "original": source_row[field],
                "exact": snapshot_rows[seq][field] == source_row[field],
            }
        write_json(
            out / "snapshot-verification.json",
            {
                "database_uri": DB_URI,
                "query": "SELECT id,course,seq,acip,wylie,english FROM corpus_segments WHERE course=? AND seq BETWEEN ? AND ? ORDER BY seq",
                "parameters": ["C05", 233, 239],
                "verified_fields": comparisons,
                "all_six_exact": all(value["exact"] for value in comparisons.values()),
            },
        )

        placeholders = ",".join("?" for _ in TERMS[seq])
        lexical_sql = (
            "SELECT id,wylie,tibetan,tibetan_source,acip,pronunciation,pronunciation_source,"
            "sanskrit,hopkins_english,hgm_gloss,tier,hgm_source,status,tenses,flags,"
            "corpus_courses,corpus_n_segments,raw FROM entries WHERE wylie IN ("
            + placeholders
            + ") ORDER BY wylie,id"
        )
        cursor = connection.execute(lexical_sql, TERMS[seq])
        lexical_rows = rows_as_dicts(cursor, cursor.fetchall())
        write_json(
            out / "glossary-originals.json",
            {
                "database_uri": DB_URI,
                "query": lexical_sql,
                "parameters": TERMS[seq],
                "requested_term_count": len(TERMS[seq]),
                "row_count": len(lexical_rows),
                "rows": lexical_rows,
            },
        )

        corpus_evidence = []
        for label, sql, base_params in CORPUS_QUERIES[seq]:
            params = list(base_params)
            if label == "exact English parallel outside C05":
                params[-1] = source_row["english"]
            cursor = connection.execute(sql, params)
            rows = rows_as_dicts(cursor, cursor.fetchall())
            corpus_evidence.append(
                {"label": label, "query": sql, "parameters": params, "row_count": len(rows), "rows": rows}
            )
        write_json(
            out / "parallel-originals.json",
            {
                "database_uri": DB_URI,
                "standing_exclusion": "Every semantic corpus query excludes C13.",
                "queries": corpus_evidence,
            },
        )

    connection.close()


if __name__ == "__main__":
    main()
