def bank(kind,record,locator):
    data=json.dumps(record,ensure_ascii=False,sort_keys=True).encode(); key=kind+'-'+hashlib.sha256(data).hexdigest()
    if key not in pool:pool[key]={'kind':kind,'record':record,'locators':[]}
    if locator not in pool[key]['locators']:pool[key]['locators'].append(locator)
    return key

def walk(x,path='$'):
    if isinstance(x,dict):
        s=json.dumps(x,ensure_ascii=False)
        if any(k in x for k in ['found','expected']) and (re.search(r'sizzle|imagination|C05:30[123]|E-071|following.{0,30}heading|E-107',s,re.I)):
            yield path,x
        else:
            for k,v in x.items():yield from walk(v,path+'.'+k)
    elif isinstance(x,list):
        for i,v in enumerate(x):yield from walk(v,path+f'[{i}]')
