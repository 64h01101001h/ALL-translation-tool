def hits(text,piece,subword=False):
    out=[];start=0
    while True:
        i=text.find(piece,start)
        if i<0:return out
        j=i+len(piece)
        if subword or ((not piece[0].isalnum() or i==0 or not text[i-1].isalnum()) and (not piece[-1].isalnum() or j==len(text) or not text[j].isalnum())):out.append([i,j])
        start=i+1
