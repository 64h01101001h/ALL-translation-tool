"""Execute the unchanged original generator with its immutable builder dependency.

Only connection opening is constrained: sqlite3.connect is routed to URI mode=ro.
No generator, resolution, emission, parser, conversion or gate logic is replaced.
"""
import sys,os,json,types,sqlite3,runpy,hashlib,subprocess
from pathlib import Path
P=Path(__file__).resolve().parent
W=P.parents[2]/'campaign-worktree'
PIN=json.loads((P.parent/'proposal-baseline.json').read_text())
COMMIT=PIN['commit']
_connect=sqlite3.connect
def connect_readonly(database,*args,**kwargs):
 path=str(database)
 if not path.startswith('file:'):path='file:'+path+'?mode=ro'
 assert 'mode=ro' in path,path
 kwargs['uri']=True
 return _connect(path,*args,**kwargs)
sqlite3.connect=connect_readonly
def load_immutable(name,rel):
 src=(P/'evidence/governance'/rel).read_bytes()
 mod=types.ModuleType(name);mod.__file__=str(W/rel);sys.modules[name]=mod
 exec(compile(src,str(W/rel),'exec'),mod.__dict__)
 return mod
def bootstrap():
 h=load_immutable('hgm_tools','engines/hgm_tools.py')
 b=load_immutable('build_alignment_layer','tools/build_alignment_layer.py')
 src=(W/'tools/gen_alignment_page.py').read_bytes()
 assert hashlib.sha256(src).hexdigest()==PIN['governing_git_pins']['tools/gen_alignment_page.py']['sha256']
 assert (P/'evidence/governance/tools/gen_alignment_page.py').read_bytes()==src
 return h,b
if __name__=='__main__':
 bootstrap()
 runpy.run_path(str(W/'tools/gen_alignment_page.py'),run_name='__main__')
