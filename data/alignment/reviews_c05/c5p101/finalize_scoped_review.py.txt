# Preserve earlier derived views, then derive current complete package without queries or generators.
derived = S/'history/needs-correction-derived-package'

package = (S/'scoped-inputs/current/semantic-input-review-package.txt').read_text()
parts = re.split(r'^===== (.*?) =====\n', package, flags=re.M)
index = []; compact = []
for name, text in zip(parts[1::2], parts[2::2]):
    body = text.strip()
    index.append({'name':name, 'bytes':len(body.encode()), 'sha256':sha(body.encode())})
    try:
        view = json.dumps(json.loads(body), ensure_ascii=False, separators=(',', ':'))
    except json.JSONDecodeError:
        view = body
    compact.append('===== '+name+' =====\n'+view+'\n')
assert len(index) == 50
dump(S/'package-section-index.json', index)
(S/'package-full-compact.txt').write_text('\n'.join(compact))
