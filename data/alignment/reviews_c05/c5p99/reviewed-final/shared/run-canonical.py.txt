import pathlib,sqlite3,runpy,sys
W=pathlib.Path(__file__).resolve().parent.parent.parent.parent/'campaign-worktree'
original=sqlite3.connect
def readonly(database,*args,**kwargs):
 if not str(database).startswith('file:'):database=pathlib.Path(database).resolve().as_uri()+'?mode=ro'
 elif 'mode=ro' not in database:raise RuntimeError('SQLite must be read-only')
 kwargs['uri']=True
 return original(database,*args,**kwargs)
sqlite3.connect=readonly
runpy.run_path(str(W/'tools/gen_alignment_page.py'),run_name='__main__')
