import json,pathlib,re
D=pathlib.Path(__file__).parent;B=D.parent;rows={r['seq']:r for r in json.loads((B/'source.json').read_text())}
def write(seq,title,note,items,omissions):
 row=rows[seq];spans=[];intended=[];cursor=0;parent=None
 for item in items:
  t,e,d,context,reason,*flags=item;sid='w'+str(len(spans)+1);sp={'id':sid,'d':d,'tib':t,'eng':e}
  if flags:sp.update(flags[0])
  if d==7:
   lo,hi=parent['tib_range'];ts=row['wylie'].index(t,lo,hi);assert row['wylie'][lo:hi].count(t)==1
   lo,hi=parent['eng_range'];es=row['english'].index(e,lo,hi)
  else:
   ts=row['wylie'].index(t,cursor);cursor=ts+len(t)
   hits=[m.start() for m in re.finditer(re.escape(context or e),row['english'])];assert len(hits)==1,(seq,t,e,context,hits)
   es=hits[0]+(context or e).index(e)
  rec={'id':sid,'d':d,'tib':t,'eng':e,'tib_range':[ts,ts+len(t)],'eng_range':[es,es+len(e)],'parent':parent['id'] if d==7 else None,'reason':reason}
  if d==5:parent=rec
  spans.append(sp);intended.append(rec)
 order=[x['id'] for x in sorted(intended,key=lambda x:x['eng_range'][0]) if x['d']!=7]
 spec={'course':'C05','segments':[{'seq':seq,'title':title,'spans':spans,'eng_order':order,'note':'PROVISIONAL machine alignment by Codex, English-first original proposal. GMR wording is preserved verbatim; no human semantic approval or hgm_gloss promotion is implied. '+note}]}
 p=D/str(seq);p.mkdir(exist_ok=True)
 for name,data in [('spec.json',spec),('intended-ranges.json',intended),('errata.json',[]),('omissions.json',omissions)]: (p/name).write_text(json.dumps(data,ensure_ascii=False,indent=2)+'\n')
